## SUMMARY
This template defines the canonical structure for all report documents in
the DLMS corpus. Reports are time-bound artefacts that record findings,
evidence, and recommendations for a specific period or event. They are
produced by CI monitor agents, metrics agents, and the SysAdmin tier and
are consumed by downstream decision agents and the Knowledge Base Agent.

### Additional_Fields

report_type:
  name:        report_type
  type:        enum
  required:    true
  nullable:    false
  position:    immediately after audit_ref
  values:
    quality:      Documents verification pass/fail rates, failure patterns,
                  and DoD breach frequency per stage.
    performance:  Documents SLA compliance, time-in-stage, and throughput.
    anomaly:      Documents detected deviations from baseline performance
                  or behaviour.
    bottleneck:   Documents stages with elevated dwell time or revision cycles.
    agent-behaviour: Documents unexpected agent action patterns detected
                  in the audit trail.
    ci-synthesis: Integrates multiple monitor reports into a unified
                  multi-perspective view.
    metrics:      General operational metrics across any scope.
    sysadmin:     SysAdmin activity and directive resolution tracking.
  constraints:
    - Immutable after first approved version
    - report_type informs which CI monitor agent produces this report
      and which downstream agent consumes it

  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset
  description: Start of the reporting period (inclusive).

period_end:
  name:        period_end
  type:        timestamp
  required:    true
  nullable:    false

  description: End of the reporting period (inclusive).
  constraints:

    - For event-triggered reports (e.g. anomaly), period_start

## REQUIRED_SECTIONS

SUMMARY:
  heading:     ## SUMMARY
  required:    true
  position:    1
  description: >
    A dense, factual statement of the report_type, the period covered,
    the top-line finding, and whether immediate action is required.
  constraints:
    - Exactly 2–3 sentences
    - Must state whether escalation was triggered (yes/no)
    - Must name the report_type and period

SCOPE:
  heading:     ## SCOPE
  required:    true
  position:    2
  description: >
    What this report covers and what is explicitly excluded.
    Defines the measurement boundary and data sources used.
  constraints:
    - At least one 'Covers:' statement
    - At least one 'Excludes:' statement
    - Must list all source artefacts consumed to produce this report

PERIOD:
  heading:     ## PERIOD
  required:    true
  position:    3
  description: >
    Structured declaration of the reporting period, document count,
    and operational context relevant to interpreting the findings.
  required_fields:
    period_start:       Timestamp (mirrors frontmatter field)
    period_end:         Timestamp (mirrors frontmatter field)
    documents_in_scope: Count of documents active during this period
    prior_period_ref:   doc_id of prior report or null
    context_notes:      >
      Any operational events during this period that may affect
      metric interpretation (e.g. bootstrap phase, SysAdmin directive
      changes, known system events). Null if none.
  constraints:
    - period_start and period_end must mirror the frontmatter values
    - documents_in_scope must be a non-negative integer

FINDINGS:
  heading:     ## FINDINGS
  required:    true
  position:    4
  description: >
    The structured findings of this report. Each finding is an
    observed fact derived from evidence. Findings do not include
    interpretation or recommendations — those are in separate sections.
  finding_schema:
    id:
      type:     string
      format:   F prefix + two-digit number (F01, F02, …)
      required: true
    category:
      type:     string
      required: true
      description: >
        The metric or observation category this finding relates to.
        Examples: verification-pass-rate, time-in-stage, sla-breach-count.
    finding:
      type:     string
      required: true
      description: >
        The observed fact. Present tense. Quantified where possible.
        Does not include cause or recommendation.
        Example: "Verification pass rate for creation stage was 87%,
        below the 95% target, for the third consecutive period."
    trend:
      type:     enum
      required: true
      values:   [improving, stable, degrading, insufficient-data]
    severity:
      type:     enum
      required: true
      values:   [info, warning, critical]
    evidence_refs:
      type:     array of strings
      required: true
      description: >
        References to EVIDENCE entries (by id) that support this finding.
  constraints:
    - Minimum 1 finding
    - Finding IDs sequential from F01 with no gaps
    - A finding with severity: critical must trigger escalation_required: true
      in the ESCALATION section
    - Findings are factual only — no interpretation or recommendation

EVIDENCE:
  heading:     ## EVIDENCE
  required:    true
  position:    5
  description: >
    The raw or aggregated data that supports the findings. Each
    evidence entry is traceable to a source artefact in the DLMS.
  evidence_schema:
    id:
      type:     string
      format:   EV prefix + two-digit number (EV01, EV02, …)
      required: true
    source_artefact:
      type:     string
      required: true
      description: >
        The file path or doc_id of the source artefact.
        Must be traceable in the audit trail.
    data_type:
      type:     string
      required: true
      description: >
        The type of data (e.g. verification-record, metrics-log,
        audit-event-count, stage-timing).
    value:
      type:     string
      required: true
      description: >
        The extracted value or data point. Quantified where possible.
    period:
      type:     string
      required: true
      description: The time period this evidence covers.
  constraints:
    - Minimum 1 evidence entry
    - Evidence IDs sequential from EV01 with no gaps
    - Every finding's evidence_refs[] must resolve to valid EV ids
    - source_artefact must reference an artefact that exists in
      the DLMS audit trail

ANALYSIS:
  heading:     ## ANALYSIS
  required:    true
  position:    6
  description: >
    Interpretation of the findings. Explains what the evidence means,
    identifies root causes or contributing factors, and establishes
    the significance of each finding. This is where the report agent
    applies reasoning — findings are facts; analysis is interpretation.
  analysis_schema:
    finding_ref:
      type:     string
      required: true
      description: The finding id (F01, F02, …) being analysed.
    interpretation:
      type:     string
      required: true
      description: >
        What this finding means in context. What is causing it?
        Is it a signal or noise? Is it isolated or part of a pattern?
    contributing_factors:
      type:     array of strings
      required: false
      description: Known or suspected causes.
    pattern_ref:
      type:     string or null
      required: false
      description: >
        If this finding matches a known pattern in the Knowledge Base,
        the doc_id of that pattern. Null if no match.
  constraints:
    - Every finding with severity: warning or critical must have
      a corresponding analysis entry
    - Info findings may be listed without analysis if self-evident

RECOMMENDATIONS:
  heading:     ## RECOMMENDATIONS
  required:    true
  position:    7
  description: >
    Structured actions recommended in response to the findings.
    Recommendations are directed at specific agents or roles.
    They are proposals — authority to act rests with the
    receiving agent or SysAdmin.
  recommendation_schema:
    id:
      type:     string
      format:   R prefix + two-digit number (R01, R02, …)
      required: true
    finding_refs:
      type:     array of strings
      required: true
      description: The finding ids that motivate this recommendation.
    recommendation:
      type:     string
      required: true
      description: >
        What is recommended. Present tense. Addressed to a specific
        agent or role. Actionable.
    directed_to:
      type:     string
      required: true
      description: >
        The agent or role this recommendation is directed to.
        Examples: agent:sla-manager, sysadmin:dlm-sysadmin,
        agent:recommendation-generator.
    priority:
      type:     enum
      required: true
      values:   [low, medium, high, immediate]
    expected_impact:
      type:     string
      required: true
      description: >
        What improvement is expected if this recommendation is acted on.
        Quantified where possible.
  constraints:
    - May be empty [] if no recommendations arise from the findings
    - If ESCALATION.escalation_required == true, at least one
      recommendation must have priority: immediate
    - IDs sequential from R01 with no gaps

ESCALATION:
  heading:     ## ESCALATION
  required:    true
  position:    8
  description: >
    Declares whether this report requires immediate SysAdmin attention
    and the basis for that determination.
  required_fields:
    escalation_required:
      type:     boolean
      description: >
        True if any finding has severity: critical OR if any metric
        has breached its critical threshold per the governing SLA.
    escalation_basis:
      type:     array of strings
      description: >
        The finding ids (F01, etc.) that trigger escalation.
        Empty array if escalation_required == false.
    escalation_target:
      type:     string or null
      description: >
        The agent or role to notify. Null if escalation_required == false.
        Must be sysadmin:dlm-sysadmin for critical escalations.
    notification_sent:
      type:     boolean
      description: >
        Whether the escalation notification has been dispatched.
        Set to true by the generating agent upon sending.
  constraints:
    - escalation_required: true requires escalation_target != null
    - escalation_required: true requires notification_sent to be
      set to true before this report enters under_review status
    - If escalation_required: false, escalation_basis must be []

CHANGE_LOG:

## VALIDATION_RULES

V01:
  rule:     All nine REQUIRED_SECTIONS must be present in prescribed order.
  test:     >
    Sections SUMMARY, SCOPE, PERIOD, FINDINGS, EVIDENCE, ANALYSIS,
    RECOMMENDATIONS, ESCALATION, CHANGE_LOG all present and in sequence.
  checked_by: Template Validator

V02:
  rule:     doc_type must be 'report'. report_type, period_start,
            period_end, generated_by, and prior_report_ref must all
            be present. period_end must be >= period_start.
  test:     >
    doc_type == 'report' AND report_type in approved enum AND
    period_start != null AND period_end != null AND
    period_end >= period_start AND generated_by != null AND
    prior_report_ref field present (value may be null)
  checked_by: Template Validator; Agent Format Validator

V03:
  rule:     generated_by must differ from verified_by.
  test:     generated_by != verified_by
  checked_by: Template Validator; Perspective Reviewer (lens-accuracy)

V04:
  rule:     FINDINGS must contain at least one entry. Every finding
            must have id, category, finding, trend, severity, and
            at least one evidence_ref.
  test:     >
    len(FINDINGS) >= 1 AND for all F:
    F.id != null AND F.category != null AND F.finding != null AND
    F.trend IN [improving, stable, degrading, insufficient-data] AND
    F.severity IN [info, warning, critical] AND
    len(F.evidence_refs) >= 1
  checked_by: Template Validator

V05:
  rule:     Every evidence_ref in FINDINGS must resolve to a valid
            EV id in EVIDENCE.
  test:     >
    for all F: for all ref in F.evidence_refs:
    ref IN set(EVIDENCE[*].id)
  checked_by: Template Validator

V06:
  rule:     Every finding with severity: warning or critical must
            have a corresponding entry in ANALYSIS.
  test:     >
    for all F where F.severity IN [warning, critical]:
    exists A in ANALYSIS where A.finding_ref == F.id
  checked_by: Template Validator

V07:
  rule:     If escalation_required is true: escalation_basis must be
            non-empty, escalation_target must be non-null, and
            notification_sent must be true before under_review status.
  test:     >
    IF ESCALATION.escalation_required == true THEN
    len(ESCALATION.escalation_basis) >= 1 AND
    ESCALATION.escalation_target != null AND
    ESCALATION.notification_sent == true
  checked_by: Template Validator; Review Orchestrator gate condition

V08:
  rule:     If any finding has severity: critical, escalation_required
            must be true.
  test:     >
    IF count(F where F.severity == critical) > 0 THEN
    ESCALATION.escalation_required == true
  checked_by: Template Validator; CI Verifier

V09:
  rule:     dependencies[] must include is-governed-by DLMS-2026-0001.
  test:     >
    dependencies contains doc_id == 'DLMS-2026-0001'
    AND rel == 'is-governed-by'
  checked_by: Dependency Index Updater; Indexing Verifier

V10:
  rule:     CHANGE_LOG must have one entry per version, newest first,
            in the prescribed format.
  test:     >
    len(CHANGE_LOG entries) == len(version history) AND
    CHANGE_LOG[0].version == current version AND
    all entries match prescribed format.
  checked_by: Storage Verifier; Version Control Agent
