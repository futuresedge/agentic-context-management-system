## SUMMARY
This template defines the canonical structure for taxonomy documents in
the DLMS corpus. There is exactly one active taxonomy document in the DLMS
at any time — the controlled vocabulary from which all tag values are drawn.
It is the authoritative lookup source for the Classification Agent,
Taxonomy Validator, and Tag Index Agent.

singleton:
  name:        singleton
  type:        boolean
  required:    true
  nullable:    false
  position:    immediately after audit_ref
  value:       true (always)
  description: >
    Declares that exactly one active (status: approved) taxonomy
    document may exist in the DLMS at any time. A new taxonomy version
    supersedes the prior one atomically — there is never a gap or
    overlap between active versions.
  constraints:
    - Always true; this field is a machine-readable guard checked by
      the Approval Gate Agent before approving any taxonomy document
    - If an approved taxonomy already exists, it must be superseded
      atomically when this version is approved

term_count:
  name:        term_count
  type:        integer
  required:    true
  nullable:    false
  position:    after singleton
  description: >
    The total count of terms in this taxonomy version with status: active.
    Deprecated terms are not counted.
  constraints:
    - Must equal the count of TERMS entries with status: active
    - Set by Author Agent; validated by Taxonomy Validator

## REQUIRED_SECTIONS

SUMMARY:
  heading:     ## SUMMARY
  required:    true
  position:    1
  description: >
    A dense statement of the taxonomy's purpose, term count, and
    the domain structure it covers.
  constraints:
    - Exactly 2–3 sentences
    - Must state the total active term count
    - Must list the top-level domains covered

SCOPE:
  heading:     ## SCOPE
  required:    true
  position:    2
  description: >
    What this taxonomy covers and what it explicitly excludes.
    Defines the semantic boundary of the controlled vocabulary.
  constraints:
    - At least one 'Applies to:' statement
    - At least one 'Excludes:' statement
    - Must state that all DLMS document tags[] must be drawn
      exclusively from this taxonomy

GOVERNANCE:
  heading:     ## GOVERNANCE
  required:    true
  position:    3
  description: >
    The rules and process for proposing, approving, and deprecating
    terms. Defines who has authority to change the taxonomy and
    what evidence is required.
  required_fields:
    proposal_process:
      description: >
        How a new term or deprecation is proposed.
        Who may propose, what artefact is produced, and where it goes.
    approval_authority:
      description: >
        Who approves taxonomy changes. Must be SysAdmin-tier.
        Taxonomy changes affect all documents with tags from
        the changed term — authority must reflect this impact.
    deprecation_rules:
      description: >
        Conditions under which a term may be deprecated rather than
        removed. A deprecated term remains in TERMS with
        status: deprecated; it may not be applied to new documents
        but existing uses remain valid until migration is complete.
    migration_process:
      description: >
        How existing documents are updated when a term is renamed
        or deprecated. Includes Tag Index Agent role and timeline.
    prohibited_terms:
      description: >
        Categories of terms that will never be added regardless of
        proposal. Examples: free-text terms, agent-internal identifiers,
        version strings.
  constraints:
    - All four fields are required and non-null
    - approval_authority must name a sysadmin role, not an agent role

DOMAINS:
  heading:     ## DOMAINS
  required:    true
  position:    4
  description: >
    The top-level organisational domains that group the taxonomy terms.
    Every term belongs to exactly one domain. Domains are not tags —
    they are organisational structure only.
  domain_schema:
    id:
      type:     string
      format:   D prefix + two-digit number (D01, D02, …)
      required: true
    name:
      type:     string
      required: true
      description: Short descriptive domain name.
    description:
      type:     string
      required: true
      description: What this domain covers. 1–2 sentences.
    term_count:
      type:     integer
      required: true
      description: Count of active terms in this domain.
  constraints:
    - Minimum 2 domains
    - Domain IDs sequential from D01 with no gaps
    - sum(domain.term_counts) must equal frontmatter term_count

TERMS:
  heading:     ## TERMS
  required:    true
  position:    5
  description: >
    The complete controlled vocabulary. Every valid tag value used
    anywhere in the DLMS must appear here with status: active.
    Terms are the atomic unit of the taxonomy.
  term_schema:
    term:
      type:     string
      required: true
      format:   lowercase-hyphen-separated-slug
      description: >
        The canonical tag value. This exact string is what appears
        in tags[] frontmatter fields across all DLMS documents.
      constraints:
        - Matches regex ^[a-z][a-z0-9-]*$
        - Unique across the entire TERMS list
        - Immutable after first approved version (rename = deprecate + add)
    domain_id:
      type:     string
      required: true
      description: The domain ID this term belongs to.
    definition:
      type:     string
      required: true
      description: >
        What this term means in the DLMS context. One precise sentence.
        Must be sufficient for Classification Agent to correctly assign
        the term without additional context.
    parent_term:
      type:     string or null
      required: true
      nullable: true
      description: >
        The term value of the parent term if this is a narrower term.
        Null if this is a top-level term within its domain.
      constraints:
        - If set, parent_term must exist in TERMS with status: active
        - Maximum hierarchy depth is 2 (parent → child; no grandchildren)
    status:
      type:     enum
      required: true
      values:
        active:     Term is in active use. May be applied to new documents.
        deprecated: Term is retained but must not be applied to new documents.
                    Existing uses remain valid until migration is complete.
    deprecated_at:
      type:     timestamp or null
      required: true
      nullable: true
      description: Timestamp of deprecation. Null if status: active.
    replaced_by:
      type:     string or null
      required: true
      nullable: true
      description: >
        The term value that replaces this one, if a replacement exists.
        Null if deprecated with no replacement.
  constraints:
    - All term values must match the slug format
    - No two terms may have the same term value
    - Parent terms must exist and be active at the time the child term
      is created
    - Deprecated terms remain in the list; they are never removed
    - Terms are ordered alphabetically within each domain

CHANGE_LOG:
  heading:     ## CHANGE_LOG
  required:    true
  position:    last
  format:      "- v[version] | [YYYY-MM-DD] | [agent_id] | [change summary]"
  constraints:
    - Newest version first; one entry per version; immutable once written
    - Change summary must state: terms added (count), terms deprecated
      (count), definitions updated (count)

## VALIDATION_RULES

V01:
  rule:     All six REQUIRED_SECTIONS must be present in prescribed order.
  test:     >
    Sections SUMMARY, SCOPE, GOVERNANCE, DOMAINS, TERMS, CHANGE_LOG
    all present and in sequence.
  checked_by: Template Validator

V02:
  rule:     doc_type must be 'taxonomy'. singleton must be true.
            term_count must equal the count of active terms in TERMS.
  test:     >
    doc_type == 'taxonomy' AND singleton == true AND
    term_count == count(T in TERMS where T.status == 'active')
  checked_by: Template Validator; Taxonomy Validator

V03:
  rule:     There must be exactly one taxonomy document with
            status: approved in the DLMS at any time.
  test:     >
    count(documents where doc_type == 'taxonomy'
    AND status == 'approved') <= 1
  checked_by: Approval Gate Agent (pre-approval check)

V04:
  rule:     All term values must match the slug format. All term
            values must be unique across the TERMS list.
  test:     >
    for all T in TERMS:
    T.term matches regex ^[a-z][a-z0-9-]*$ AND
    T.term is unique in TERMS
  checked_by: Taxonomy Validator; Template Validator

V05:
  rule:     Every term must reference a valid domain_id from DOMAINS.
            sum of per-domain term counts must equal frontmatter term_count.
  test:     >
    for all T in TERMS where T.status == 'active':
    T.domain_id IN set(DOMAINS[*].id) AND
    sum(D.term_count for D in DOMAINS) == term_count
  checked_by: Taxonomy Validator

V06:
  rule:     Parent terms must exist in TERMS with status: active.
            Hierarchy depth must not exceed 2.
  test:     >
    for all T where T.parent_term != null:
    T.parent_term IN set(TERMS[*].term where status == 'active') AND
    referenced parent_term has parent_term == null
  checked_by: Taxonomy Validator

V07:
  rule:     Deprecated terms must have a non-null deprecated_at timestamp.
            Active terms must have deprecated_at == null.
  test:     >
    for all T where T.status == 'deprecated': T.deprecated_at != null AND
    for all T where T.status == 'active': T.deprecated_at == null
  checked_by: Taxonomy Validator

V08:
  rule:     GOVERNANCE must define proposal_process, approval_authority,
            deprecation_rules, migration_process, and prohibited_terms.
            All fields must be non-null.
  test:     >
    all five GOVERNANCE fields are present and non-null AND
    approval_authority references a sysadmin role
  checked_by: Template Validator

V09:
  rule:     CHANGE_LOG must have one entry per version, newest first,
            stating terms added, deprecated, and updated counts.
  test:     >
    len(CHANGE_LOG entries) == len(version history) AND
    CHANGE_LOG[0].version == current version AND
    all entries match prescribed format AND
    each entry references added/deprecated/updated counts
  checked_by: Storage Verifier; Version Control Agent
