## SUMMARY
This template defines the canonical structure for all reference documents
in the DLMS corpus. Reference documents are structured lookup artefacts —
catalogues, registries, schedules, and controlled vocabularies — consumed
directly by agents as authoritative data sources rather than read as prose.

### Additional_Fields

reference_type:
  name:        reference_type
  type:        enum
  required:    true
  nullable:    false
  position:    immediately after audit_ref
  values:
    catalogue:   An enumerated list of named items with structured
                 properties. Example: agent registry, doc_type catalogue.
    registry:    A live record of current system state. Entries are
                 added as items are created and updated as state changes.
                 Example: document registry, version registry.
    schedule:    A time- or condition-ordered reference.
                 Example: retention schedule, review schedule.
    lookup:      A key-value or key-object reference consumed by agents
                 for field validation. Example: retention class definitions,
                 taxonomy.
  constraints:
    - Immutable after first approved version
    - Informs which index agent maintains this reference

entry_schema_ref:
  name:        entry_schema_ref
  type:        string or null
  required:    true
  nullable:    true
  position:    after reference_type
  format:      doc_id or null
  description: >
    If this reference document's entries conform to a schema defined
    in another DLMS document (e.g. a specification), that doc_id
    is recorded here.
  constraints:
    - If set, must appear in dependencies[] with rel: references
    - Null if entry schema is fully defined within this document

maintenance_owner:
  name:        maintenance_owner
  type:        string
  required:    true
  nullable:    false
  position:    after entry_schema_ref
  format:      agent:[agent-role-slug] OR sysadmin:[role-slug]
  description: >
    The agent or role accountable for keeping this reference current.
    Responsible for initiating updates when entries become stale.
  constraints:
    - Must reference an entry in /dlms/registry/agent-registry.md

update_trigger:
  name:        update_trigger
  type:        string
  required:    true

    - "A regulatory change affects retention requirements."

## REQUIRED_SECTIONS

SUMMARY:
  heading:     ## SUMMARY
  required:    true
  position:    1
  description: >
    A dense, factual statement of what this reference contains,
    its reference_type, and which agents consume it.
  constraints:
    - Exactly 2–3 sentences
    - Must name the reference_type and primary consumers

SCOPE:
  heading:     ## SCOPE
  required:    true
  position:    2
  description: >
    What this reference covers and what it explicitly excludes.
    Defines the boundaries of the lookup space.
  constraints:
    - At least one 'Applies to:' statement
    - At least one 'Excludes:' statement
    - Must state whether entries are exhaustive (closed set) or
      extensible (open set with governance process)

ENTRY_SCHEMA:
  heading:     ## ENTRY_SCHEMA
  required:    true
  position:    3
  description: >
    The structure of each entry in this reference document.
    Defines all fields, their types, and constraints for a single
    entry. This is the schema an agent uses to parse ENTRIES.
  field_schema:
    field_name:
      type:     string
      required: true
    field_type:
      type:     string
      required: true
    required:
      type:     boolean
      required: true
    nullable:
      type:     boolean
      required: true
    description:
      type:     string
      required: true
    constraints:
      type:     array of strings
      required: false
  constraints:
    - Minimum 2 fields per entry schema
    - Every field in ENTRIES must be declared here
    - Fields not declared here must not appear in ENTRIES

ENTRIES:
  heading:     ## ENTRIES
  required:    true
  position:    4
  description: >
    The actual reference data. Each entry conforms to ENTRY_SCHEMA.
    Entries are structured for machine consumption — consistent
    keys, predictable values, no prose within entry fields.
  constraints:
    - All entries must conform to ENTRY_SCHEMA
    - Entries are ordered by the primary key field (defined in ENTRY_SCHEMA)
    - For reference_type: registry, entries reflect current state;
      historical states are in the version history, not inline
    - For reference_type: lookup, every valid lookup key must have an entry
    - Empty ENTRIES (no entries yet) must state: "No entries. Reference
      is initialised and awaiting first population."

MAINTENANCE:
  heading:     ## MAINTENANCE
  required:    true
  position:    5
  description: >
    How entries are added, updated, and retired. Specifies the
    process and authority for each operation type.
  operation_schema:
    operation:
      type:     enum
      values:   [add, update, retire]
      required: true
    authority:
      type:     string
      required: true
      description: Agent or role that authorises this operation.
    process:
      type:     string
      required: true
      description: >
        The steps taken to perform this operation. May reference
        a procedure doc_id if a full procedure exists.
    audit_requirement:
      type:     string
      required: true
      description: What must be logged when this operation is performed.
  constraints:
    - All three operations (add, update, retire) must be specified
    - Authority for retire operations must be at least as strict
      as for add operations
    - All operations must produce an audit event captured by
      Audit Trail Agent

REVIEW_SCHEDULE:
  heading:     ## REVIEW_SCHEDULE
  required:    true
  position:    6
  description: Maximum review interval and unscheduled review triggers.
  required_fields:
    interval_max:       6 months maximum (Tier 4 document)
    trigger_conditions: Minimum 2 conditions; at least one CI-related.

CHANGE_LOG:
  heading:     ## CHANGE_LOG
  required:    true
  position:    last
  format:      "- v[version] | [YYYY-MM-DD] | [agent_id] | [change summary]"
  constraints:
    - Newest version first; one entry per version; immutable once written
    - Change summary must state which entries were added, updated,
      or retired and the count of changes
