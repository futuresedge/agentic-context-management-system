## SUMMARY
This template defines the canonical structure for all SLA documents in the
DLMS corpus. SLA documents specify measurable performance targets for DLMS
stages and services, defining the metrics tracked, how they are measured,
breach thresholds, and the response process when thresholds are exceeded.

sla_type:
  name:        sla_type
  type:        enum
  required:    true
  nullable:    false
  position:    immediately after audit_ref
  values:
    stage:     Performance targets for a specific DLMS lifecycle stage.
    delivery:  Performance targets for the Context Delivery Layer.
    ci:        Performance targets for the CI monitoring cycle.
    sysadmin:  Performance targets for SysAdmin directive processing.
  constraints:
    - Immutable after first approved version
    - One SLA document per sla_type per governed scope

  name:        governed_scope

    - [creation, review, indexing]
    - [context-delivery-layer]
    - [ci-cycle]
  constraints:
    - At least one entry required
    - Values must reference named stages or services in the DLMS registry

## REQUIRED_SECTIONS

SUMMARY:
  heading:     ## SUMMARY
  required:    true
  position:    1
  description: >
    A dense, factual statement of what this SLA governs, the sla_type,
    and the primary performance dimension it targets.
  constraints:
    - Exactly 2–3 sentences
    - Must name the governed_scope and primary metric category

SCOPE:
  heading:     ## SCOPE
  required:    true
  position:    2
  description: >
    What this SLA applies to and what it explicitly excludes.
    Identifies the measurement boundary.
  constraints:
    - At least one 'Applies to:' statement
    - At least one 'Excludes:' statement
    - Boundary conditions (e.g. clock starts when X, stops when Y)
      must be declared here

DEFINITIONS:
  heading:     ## DEFINITIONS
  required:    true
  position:    3
  description: >
    Definitions of measurement terms, units, and threshold concepts
    used in this SLA. Alphabetical.
  constraints:
    - Must define every unit used in METRICS
    - Must define 'breach' as it applies to this SLA specifically
    - If no domain-specific terms: state "No domain-specific definitions."

METRICS:
  heading:     ## METRICS
  required:    true
  position:    4
  description: >
    The enumerated performance metrics this SLA governs. Each metric
    is independently measurable with a defined target and breach threshold.
  metric_schema:
    id:
      type:     string
      format:   M prefix + two-digit number (M01, M02, …)
      required: true
    name:
      type:     string
      required: true
      description: Short descriptive name for the metric.
    description:
      type:     string
      required: true
      description: What this metric measures and why it matters.
    unit:
      type:     string
      required: true
      description: The unit of measurement (seconds, percentage, count, etc.)
    target:
      type:     string
      required: true
      description: >
        The performance level that constitutes compliance.
        Expressed as: operator + value + unit.
        Examples: "<= 30 seconds", ">= 99%", "== 0 breaches"
    warning_threshold:
      type:     string
      required: true
      description: >
        The level at which a warning is triggered before a breach occurs.
        Expressed in same format as target.
    breach_threshold:
      type:     string
      required: true
      description: >
        The level at which an SLA breach is declared.
        Expressed in same format as target.
    measurement_source:
      type:     string
      required: true
      description: >
        Which agent or log produces the raw data for this metric.
        Must reference a specific agent or artefact type.
    measurement_frequency:
      type:     string
      required: true
      description: >
        How often this metric is sampled.
        Examples: per-document, per-stage-completion, hourly, per-CI-cycle
  constraints:
    - Minimum 1 metric; no maximum
    - IDs sequential from M01 with no gaps
    - target must be strictly more permissive than breach_threshold
    - warning_threshold must sit between target and breach_threshold
    - All measurement_source values must reference agents or artefacts
      that exist in the DLMS registry

MEASUREMENT_METHOD:
  heading:     ## MEASUREMENT_METHOD
  required:    true
  position:    5
  description: >
    How metrics are collected, aggregated, and reported. Describes
    the end-to-end measurement pipeline from raw event to reported value.
  required_fields:
    collection_agent:    >
      The agent responsible for collecting raw measurement data.
      Typically a per-stage Metrics Agent.
    aggregation_agent:   >
      The agent responsible for aggregating raw data into reported
      metric values. Typically Metrics Aggregator.
    reporting_artefact:  >
      The artefact that carries reported metric values.
      Must reference a named TMPL-[report]-001 template.
    reporting_frequency: >
      How often the reported metric values are produced and reviewed.
    baseline_ref:        >
      How the performance baseline is established and stored.
      The Anomaly Detector compares live metrics against this baseline.
  constraints:
    - collection_agent and aggregation_agent must differ
      (Verification Independence Policy applies)
    - reporting_artefact must reference an approved template

BREACH_RESPONSE:
  heading:     ## BREACH_RESPONSE
  required:    true
  position:    6
  description: >
    The structured response process activated when a metric reaches
    breach_threshold. Defines notification, escalation, and resolution.
  response_schema:
    level:
      type:     enum
      values:   [warning, breach, critical]
      required: true
    trigger:
      type:     string
      required: true
      description: >
        The metric condition that activates this response level.
        References a metric id and threshold.
    notification_targets:
      type:     array of strings
      required: true
      description: Agents or roles notified at this level.
    response_actions:
      type:     array of strings
      required: true
      description: >
        Ordered actions taken in response. Each action names
        the responsible agent and what it does.
    resolution_sla:
      type:     string
      required: true
      description: >
        How long the responding team has to resolve or escalate
        before the next level is triggered.
  constraints:
    - All three levels (warning, breach, critical) must be defined
    - notification_targets at breach level must include
      agent:sla-manager
    - notification_targets at critical level must include
      sysadmin:dlm-sysadmin
    - resolution_sla at critical level must be the shortest

REVIEW_SCHEDULE:
  heading:     ## REVIEW_SCHEDULE
  required:    true
  position:    7
  description: Maximum review interval and unscheduled review triggers.
  required_fields:
    interval_max:       3 months maximum (Tier 5 document)
    trigger_conditions: Minimum 2 conditions; at least one SLA-breach-related.
  constraints:
    - interval_max must not exceed 3 months

CHANGE_LOG:
  heading:     ## CHANGE_LOG
  required:    true
  position:    last
  format:      "- v[version] | [YYYY-MM-DD] | [agent_id] | [change summary]"
  constraints:
    - Newest version first; one entry per version; immutable once written
    - Change summary must state which metrics or thresholds changed
      and the direction of change (tightened/relaxed)

EXCLUSIONS:

  description: >

    and must not be counted as breaches.
  position:    After NOTES if present, otherwise after CHANGE_LOG
