## SUMMARY
This template defines the schema and structure of a bottleneck report
artefact produced by the Bottleneck Analyst at the end of each CI cycle.
It derives stage-level bottleneck scores from the performance report,
ranks all stages by composite dwell, revision, and verifier-failure
indicators, and surfaces the highest-impact constraints for routing to
the CI Perspective Synthesizer.

## TARGET_DOC_TYPE
bottleneck-report

## SCHEMA

### Fields

FIELD_01:
  name:        report_id
  type:        string
  required:    true
  nullable:    false
  format:      BNR-{YYYY}-{NNNN}
  example:     BNR-2026-0001
  description: Unique identifier for this bottleneck report. NNNN is a
               zero-padded sequential integer scoped to the year.
               Immutable after assignment.

FIELD_02:
  name:        cycle_id
  type:        string
  required:    true
  nullable:    false
  format:      CYCLE-{YYYY}-{NNNN}

FIELD_03:
  name:        generated_by
  type:        string
  required:    true
  nullable:    false
  format:      agent:[agent-role-slug]
  constraints:
    - Must be agent:bottleneck-analyst
    - Must differ from verified_by (Verification Independence Policy
      DLMS-2026-0014)

FIELD_04:
  name:        period_start
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset

FIELD_05:
  name:        period_end
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset
  constraints:
    - period_end must be >= period_start

FIELD_06:
  name:        prior_report_ref
  type:        string or null
  required:    true
  nullable:    true
  description: report_id of the prior bottleneck report. Null for first report.

FIELD_07:
  name:        source_performance_report_ref
  type:        string
  required:    true
  nullable:    false
  description: >
    The report_id of the performance report for the same CI cycle.
    Must be a PR-{YYYY}-{NNNN} identifier. This is the primary input
    to the Bottleneck Analyst.

FIELD_08:
  name:        stage_analysis
  type:        array of objects
  required:    true
  nullable:    false
  description: >
    Stage-level bottleneck analysis. One entry per stage present in the
    source performance report's stages_covered array.
  item_schema:
    stage:
      type:        string
      description: Stage name matching an entry in the performance report.
    avg_dwell_time_hours:
      type:        number
      description: Mean time documents spent in this stage this cycle.
    p95_dwell_time_hours:
      type:        number
      description: 95th-percentile dwell time for this stage this cycle.
    revision_rate_pct:
      type:        number
      description: >
        Proportion of documents in this stage that required revision or
        resubmission. Expressed as decimal 0–1.
    verifier_fail_rate_pct:
      type:        number
      description: >
        Proportion of verifications in this stage that produced a 'fail'
        verdict. Expressed as decimal 0–1.
    bottleneck_score:
      type:        number
      description: >
        Composite score combining avg_dwell_time_hours, revision_rate_pct,
        and verifier_fail_rate_pct with weights determined by the
        Bottleneck Analyst at runtime. Higher score indicates greater
        bottleneck severity. Scale is not normalised — used for ranking
        only.
    rank:
      type:        integer
      description: >
        Rank of this stage from most bottlenecked (1) to least. Assigned
        after all stages are scored. Sequential with no gaps.

FIELD_09:
  name:        top_bottleneck
  type:        string or null
  required:    true
  nullable:    true
  description: >
    Stage name of the stage with rank == 1. Null when no stage exceeds
    the minimum bottleneck_score threshold to warrant classification as
    a bottleneck (all stages within normal range).

FIELD_10:
  name:        bottleneck_count
  type:        integer
  required:    true
  nullable:    false
  description: >
    Count of stages whose bottleneck_score exceeds the agent-defined
    threshold for bottleneck classification. May be zero.
  constraints:
    - Must be >= 0
    - When 0, top_bottleneck must be null

## REQUIRED_SECTIONS

SUMMARY:
  heading:     ## SUMMARY
  required:    true
  position:    1
  constraints:
    - Exactly 2–3 sentences
    - Must name top_bottleneck (or state 'no bottleneck detected')
    - Must state bottleneck_count and whether escalation was triggered

SCOPE:
  heading:     ## SCOPE
  required:    true
  position:    2
  constraints:
    - Must name the source performance report by ID
    - At least one Covers: statement, one Excludes: statement

PERIOD:
  heading:     ## PERIOD
  required:    true
  position:    3
  required_fields:
    period_start:                  Timestamp — mirrors FIELD_04
    period_end:                    Timestamp — mirrors FIELD_05
    source_performance_report_ref: Report ID — mirrors FIELD_07
    prior_report_ref:              Report ID or null — mirrors FIELD_06
    context_notes:                 Null if none

FINDINGS:
  heading:     ## FINDINGS
  required:    true
  position:    4
  description: >
    One finding per stage classified as a bottleneck (bottleneck_score
    above threshold). When bottleneck_count == 0, one explicit info
    finding stating no bottlenecks detected.
  finding_schema:
    id:            string — F prefix + two-digit sequential number
    category:      string — stage name
    finding:       string — dwell time, revision rate, verifier fail rate fact
    trend:         enum — [improving, stable, degrading, insufficient-data]
    severity:      enum — [info, warning, critical]
    evidence_refs: array of strings — EV ids
  constraints:
    - F ids sequential from F01 with no gaps
    - The top-ranked bottleneck stage always yields severity: warning
      minimum; severity: critical when p95_dwell_time_hours exceeds
      2× the SLA breach_threshold for that stage

EVIDENCE:
  heading:     ## EVIDENCE
  required:    true
  position:    5
  evidence_schema:
    id:              string — EV prefix + two-digit number
    source_artefact: string — performance report ID
    data_type:       enum — [metrics-log, stage-timing, verifier-fail-rate]
    value:           string — extracted metric value
    period:          string — time period covered
  constraints:
    - Minimum 1 entry
    - EV ids sequential from EV01 with no gaps
    - Every finding's evidence_refs must resolve to valid EV ids

ANALYSIS:
  heading:     ## ANALYSIS
  required:    true
  position:    6
  analysis_schema:
    finding_ref:          string — F id
    interpretation:       string — what the bottleneck_score means; root cause
    contributing_factors: array of strings — optional
    pattern_ref:          string or null
  constraints:
    - Every finding with severity: warning or critical requires an entry
    - Must note whether the rank ordering changed from prior_report_ref
      when prior_report_ref is non-null

RECOMMENDATIONS:
  heading:     ## RECOMMENDATIONS
  required:    true
  position:    7
  recommendation_schema:
    id:              string — R prefix + two-digit number
    finding_refs:    array of strings — F ids
    recommendation:  string — present tense, actionable
    directed_to:     string — agent:[slug] or sysadmin:dlm-sysadmin
    priority:        enum — [low, medium, high, immediate]
    expected_impact: string
  constraints:
    - May be empty [] if bottleneck_count == 0
    - If escalation_required == true, at least one recommendation with
      priority: immediate

ESCALATION:
  heading:     ## ESCALATION
  required:    true
  position:    8
  required_fields:
    escalation_required: >
      Boolean. True if any finding has severity: critical (p95 dwell
      exceeds 2× SLA breach threshold).
    escalation_basis:    Array of F ids. Empty [] when escalation_required == false.
    escalation_target:   sysadmin:dlm-sysadmin or null.
    notification_sent:   Boolean.

CHANGE_LOG:
  heading:     ## CHANGE_LOG
  required:    true
  position:    last
  format:      "- v[version] | [YYYY-MM-DD] | [agent_id] | [change summary]"

## OPTIONAL_SECTIONS

NOTES:
  heading:      ## NOTES
  description:  Commentary on bottleneck scoring methodology or threshold
                calibration.
  when_to_include: Bootstrap phase or agent-version-change methodology notes.
  position:     After CHANGE_LOG

## VALIDATION_RULES

V01:
  rule:     report_id is non-null, matches BNR-{YYYY}-{NNNN}, and is unique
            for the year.
  test:     >
    report_id != null AND
    report_id matches regex ^BNR-\d{4}-\d{4}$ AND
    no other bottleneck report for the same year shares this id.
  checked_by: CI Cycle Orchestrator; Audit Trail Agent

V02:
  rule:     generated_by must be agent:bottleneck-analyst.
  test:     generated_by == "agent:bottleneck-analyst"
  checked_by: CI Cycle Orchestrator

V03:
  rule:     period_end must be >= period_start.
  test:     period_end >= period_start
  checked_by: CI Cycle Orchestrator

V04:
  rule:     stage_analysis must contain exactly one entry per stage in the
            source performance report's stages_covered. Ranks must be
            sequential from 1 with no gaps.
  test:     >
    len(stage_analysis) == len(source_performance_report.stages_covered) AND
    sorted([S.rank for S in stage_analysis]) == list(range(1, len+1))
  checked_by: Agent Format Validator; Bottleneck Analyst

V05:
  rule:     top_bottleneck must be the stage with rank == 1 when
            bottleneck_count > 0, and must be null when bottleneck_count == 0.
  test:     >
    IF bottleneck_count > 0
      THEN top_bottleneck == stage_analysis[rank==1].stage
    ELSE top_bottleneck == null
  checked_by: Agent Format Validator; Bottleneck Analyst

V06:
  rule:     bottleneck_count must equal the count of stage_analysis entries
            whose bottleneck_score exceeds the agent-defined threshold.
  test:     >
    bottleneck_count == len([S for S in stage_analysis
                             where S.bottleneck_score > bottleneck_threshold])
  checked_by: Bottleneck Analyst; Agent Format Validator

V07:
  rule:     All nine required sections must be present and appear in the
            prescribed order.
  test:     >
    Sections SUMMARY, SCOPE, PERIOD, FINDINGS, EVIDENCE, ANALYSIS,
    RECOMMENDATIONS, ESCALATION, CHANGE_LOG all present and in sequence.
  checked_by: Agent Format Validator; Template Validator
