## SUMMARY
This template defines the schema and structure of an agent behaviour report
artefact produced by the Agent Behaviour Analyst at the end of each CI cycle.
It records unexpected or non-compliant patterns detected in the audit trail —
including context boundary violations, retry exceedances, and out-of-scope
reads — as governed by the Audit Trail Policy (DLMS-2026-0015) and the Agent
Context Boundary Policy (DLMS-2026-0016).

## TARGET_DOC_TYPE
agent-behaviour-report

## SCHEMA

### Fields

FIELD_01:
  name:        report_id
  type:        string
  required:    true
  nullable:    false
  format:      ABR-{YYYY}-{NNNN}
  example:     ABR-2026-0001
  description: Unique identifier for this agent behaviour report. NNNN is a
               zero-padded sequential integer scoped to the year.
               Immutable after assignment.

FIELD_02:
  name:        cycle_id
  type:        string
  required:    true
  nullable:    false
  format:      CYCLE-{YYYY}-{NNNN}

FIELD_03:
  name:        generated_by
  type:        string
  required:    true
  nullable:    false
  format:      agent:[agent-role-slug]
  constraints:
    - Must be agent:agent-behaviour-analyst
    - Must differ from verified_by (Verification Independence Policy
      DLMS-2026-0014)

FIELD_04:
  name:        period_start
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset

FIELD_05:
  name:        period_end
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset
  constraints:
    - period_end must be >= period_start

FIELD_06:
  name:        prior_report_ref
  type:        string or null
  required:    true
  nullable:    true
  description: report_id of the prior agent behaviour report. Null for first.

FIELD_07:
  name:        audit_events_analysed
  type:        integer
  required:    true
  nullable:    false
  description: Total count of audit trail events analysed for this cycle.
  constraints:
    - Must be >= 0

FIELD_08:
  name:        agents_assessed
  type:        array of strings
  required:    true
  nullable:    false
  description: >
    List of agent_id values whose audit trail events were analysed during
    this cycle. Each entry must match an entry in the agent registry.
  constraints:
    - Minimum 1 entry

FIELD_09:
  name:        behaviour_findings
  type:        array of objects
  required:    true
  nullable:    false
  description: >
    Structured records of detected behavioural anomalies or violations.
    Empty array [] when no issues were detected. Each entry describes
    one pattern observed across one or more events for a specific agent.
  item_schema:
    id:
      type:        string
      format:      BF-{NN}
      description: Sequential identifier from BF-01; no gaps.
    agent_id:
      type:        string
      description: The agent whose behaviour triggered this finding.
    pattern_type:
      type:        enum
      values:      [context-overrun, out-of-scope-read, retry-exceeded,
                   unexpected-write, other]
      description: >
        context-overrun: agent consumed context exceeding its declared
                         token budget per DLMS-2026-0016.
        out-of-scope-read: agent read a resource outside its declared
                           READS list per DLMS-2026-0016.
        retry-exceeded:  agent retried an operation beyond permitted
                         attempt count.
        unexpected-write: agent wrote to a location outside its declared
                          WRITES target.
        other:           Pattern not covered by the above categories;
                         description field required.
    instance_count:
      type:        integer
      description: Number of audit events exhibiting this pattern.
    description:
      type:        string
      description: >
        Specific description of the observed behaviour. Required when
        pattern_type is 'other'. For other types, adds detail beyond
        the category label.
    severity:
      type:        enum
      values:      [info, warning, critical]
      description: >
        info:     isolated instance; within tolerable range.
        warning:  recurring pattern (≥3 instances) or single Tier-2
                  boundary violation.
        critical: Tier-1 or Tier-2 policy violation per DLMS-2026-0016
                  that requires immediate SysAdmin review.

FIELD_10:
  name:        violation_count
  type:        integer
  required:    true
  nullable:    false
  description: >
    Count of behaviour_findings with severity: warning or critical.
    Excludes severity: info findings.
  constraints:
    - Must equal len([BF for BF in behaviour_findings
                      where BF.severity IN ["warning", "critical"]])
    - Must be >= 0

## REQUIRED_SECTIONS

SUMMARY:
  heading:     ## SUMMARY
  required:    true
  position:    1
  constraints:
    - Exactly 2–3 sentences
    - Must state violation_count and total behaviour_findings count
    - Must state whether escalation was triggered (yes/no)
    - Must name the cycle and audit_events_analysed count

SCOPE:
  heading:     ## SCOPE
  required:    true
  position:    2
  constraints:
    - Must state the source of audit trail events (filtered slice only,
      per DLMS-2026-0016 Agent Context Boundary Policy)
    - Must list agents_assessed
    - At least one Covers: statement and one Excludes: statement

PERIOD:
  heading:     ## PERIOD
  required:    true
  position:    3
  required_fields:
    period_start:          Timestamp — mirrors FIELD_04
    period_end:            Timestamp — mirrors FIELD_05
    audit_events_analysed: Integer — mirrors FIELD_07
    agents_assessed:       Array — mirrors FIELD_08
    prior_report_ref:      Report ID or null — mirrors FIELD_06
    context_notes:         Null if none

FINDINGS:
  heading:     ## FINDINGS
  required:    true
  position:    4
  description: >
    One finding per behaviour_findings entry, grouped by agent where
    multiple findings exist for the same agent. When violation_count == 0
    and behaviour_findings is empty, one explicit info finding stating
    no violations detected.
  finding_schema:
    id:            string — F prefix + two-digit sequential number
    category:      string — agent_id + pattern_type
    finding:       string — present tense, quantified statement of pattern
    trend:         enum — [improving, stable, degrading, insufficient-data]
    severity:      enum — [info, warning, critical]
    evidence_refs: array of strings — EV ids
  constraints:
    - F ids sequential from F01 with no gaps
    - Each BF-{NN} entry in behaviour_findings must correspond to
      at least one finding in FINDINGS

EVIDENCE:
  heading:     ## EVIDENCE
  required:    true
  position:    5
  evidence_schema:
    id:              string — EV prefix + two-digit number
    source_artefact: string — audit trail log reference or event ID range
    data_type:       enum — [audit-event, audit-trail-slice, agent-log]
    value:           string — count or description of events observed
    period:          string — time period covered
  constraints:
    - Minimum 1 entry
    - EV ids sequential from EV01 with no gaps
    - Every finding's evidence_refs must resolve to valid EV ids

ANALYSIS:
  heading:     ## ANALYSIS
  required:    true
  position:    6
  analysis_schema:
    finding_ref:          string — F id
    interpretation:       string — significance; context boundary implications
    contributing_factors: array of strings — optional
    pattern_ref:          string or null
  constraints:
    - Every finding with severity: warning or critical requires an entry
    - Must cite the specific DLMS-2026-0016 rule breached (by rule id)
      for any out-of-scope-read or unexpected-write finding

RECOMMENDATIONS:
  heading:     ## RECOMMENDATIONS
  required:    true
  position:    7
  recommendation_schema:
    id:              string — R prefix + two-digit number
    finding_refs:    array of strings — F ids
    recommendation:  string — present tense, actionable
    directed_to:     string — agent:[slug] or sysadmin:dlm-sysadmin
    priority:        enum — [low, medium, high, immediate]
    expected_impact: string
  constraints:
    - May be empty [] if violation_count == 0
    - If escalation_required == true, at least one recommendation with
      priority: immediate

ESCALATION:
  heading:     ## ESCALATION
  required:    true
  position:    8
  required_fields:
    escalation_required: >
      Boolean. True if any behaviour_finding has severity: critical
      (Tier-1/Tier-2 policy violation per DLMS-2026-0016).
    escalation_basis:    >
      Array of BF ids triggering escalation. Empty [] when
      escalation_required == false.
    escalation_target:   sysadmin:dlm-sysadmin or null.
    notification_sent:   Boolean. True once notification dispatched.

CHANGE_LOG:
  heading:     ## CHANGE_LOG
  required:    true
  position:    last
  format:      "- v[version] | [YYYY-MM-DD] | [agent_id] | [change summary]"

## OPTIONAL_SECTIONS

NOTES:
  heading:      ## NOTES
  description:  Non-normative notes on audit trail coverage or boundary
                policy interpretation.
  when_to_include: >
    Agent registry changes mid-cycle; known audit-trail gaps; detection
    methodology updates.
  position:     After CHANGE_LOG

## VALIDATION_RULES

V01:
  rule:     report_id is non-null, matches ABR-{YYYY}-{NNNN}, and is unique
            for the year.
  test:     >
    report_id != null AND
    report_id matches regex ^ABR-\d{4}-\d{4}$ AND
    no other agent behaviour report for the same year shares this id.
  checked_by: CI Cycle Orchestrator; Audit Trail Agent

V02:
  rule:     generated_by must be agent:agent-behaviour-analyst.
  test:     generated_by == "agent:agent-behaviour-analyst"
  checked_by: CI Cycle Orchestrator

V03:
  rule:     period_end must be >= period_start.
  test:     period_end >= period_start
  checked_by: CI Cycle Orchestrator

V04:
  rule:     All BF-{NN} ids in behaviour_findings must be sequential from
            BF-01 with no gaps.
  test:     >
    for i, BF in enumerate(behaviour_findings):
      BF.id == "BF-" + zero_pad(i+1, 2)
  checked_by: Agent Format Validator

V05:
  rule:     violation_count must equal the count of behaviour_findings with
            severity: warning or critical.
  test:     >
    violation_count ==
      len([BF for BF in behaviour_findings
           where BF.severity IN ["warning", "critical"]])
  checked_by: Agent Format Validator; Agent Behaviour Analyst

V06:
  rule:     escalation_required must be true when any behaviour_finding has
            severity: critical.
  test:     >
    IF any behaviour_findings[*].severity == "critical"
    THEN escalation_required == true
  checked_by: CI Cycle Orchestrator; Audit Trail Agent

V07:
  rule:     Any finding with pattern_type: out-of-scope-read or
            unexpected-write must have its corresponding ANALYSIS entry
            cite the specific DLMS-2026-0016 rule identifier breached.
  test:     >
    For all BF where BF.pattern_type IN
      ["out-of-scope-read", "unexpected-write"]:
    EXISTS A in ANALYSIS where A.finding_ref maps to BF's F id AND
    A.interpretation contains a DLMS-2026-0016 rule reference
  checked_by: Agent Format Validator; CI Perspective Synthesizer

V08:
  rule:     All nine required sections must be present and appear in the
            prescribed order.
  test:     >
    Sections SUMMARY, SCOPE, PERIOD, FINDINGS, EVIDENCE, ANALYSIS,
    RECOMMENDATIONS, ESCALATION, CHANGE_LOG all present and in sequence.
  checked_by: Agent Format Validator; Template Validator
