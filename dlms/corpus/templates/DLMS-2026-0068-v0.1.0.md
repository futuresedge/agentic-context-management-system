## SUMMARY
This template defines the schema and structure of an anomaly report artefact
produced by the Anomaly Detector at the end of each CI cycle. It captures
deviations from the operational baseline that exceed detection thresholds,
sourcing data from the quality report and performance report for the same
cycle, and is consumed by the CI Perspective Synthesizer and — when critical
anomalies are detected — by SysAdmin directly.

## TARGET_DOC_TYPE
anomaly-report

## SCHEMA

### Fields

FIELD_01:
  name:        report_id
  type:        string
  required:    true
  nullable:    false
  format:      AR-{YYYY}-{NNNN}
  example:     AR-2026-0001
  description: Unique identifier for this anomaly report. NNNN is a
               zero-padded sequential integer scoped to the year.
               Immutable after assignment.

FIELD_02:
  name:        cycle_id
  type:        string
  required:    true
  nullable:    false
  format:      CYCLE-{YYYY}-{NNNN}
  description: The CI cycle this report belongs to.

FIELD_03:
  name:        generated_by
  type:        string
  required:    true
  nullable:    false
  format:      agent:[agent-role-slug]
  constraints:
    - Must be agent:anomaly-detector
    - Must differ from verified_by (Verification Independence Policy
      DLMS-2026-0014)

FIELD_04:
  name:        period_start
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset

FIELD_05:
  name:        period_end
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset
  constraints:
    - period_end must be >= period_start
    - For event-triggered anomalies, period_start and period_end
      may be identical

FIELD_06:
  name:        prior_report_ref
  type:        string or null
  required:    true
  nullable:    true
  description: >
    The report_id of the most recent prior anomaly report. Null for the
    first report.

FIELD_07:
  name:        source_quality_report_ref
  type:        string
  required:    true
  nullable:    false
  description: >
    The report_id of the quality report for the same CI cycle. The
    Anomaly Detector reads this artefact as a primary input source.
    Must be a QR-{YYYY}-{NNNN} identifier.

FIELD_08:
  name:        source_performance_report_ref
  type:        string
  required:    true
  nullable:    false
  description: >
    The report_id of the performance report for the same CI cycle. The
    Anomaly Detector reads this artefact as a primary input source.
    Must be a PR-{YYYY}-{NNNN} identifier.

FIELD_09:
  name:        baseline_ref
  type:        string or null
  required:    true
  nullable:    true
  description: >
    Reference to the operational baseline document used for deviation
    comparison. Null during the first 90 days of live operation (bootstrap
    window) when no operational baseline has been established. Once null,
    context_notes in PERIOD must acknowledge the bootstrap status.
  constraints:
    - After 90-day bootstrap window, this field must be non-null
    - The referenced baseline must be the Metrics Aggregator's current
      operational baseline document

FIELD_10:
  name:        anomaly_count
  type:        integer
  required:    true
  nullable:    false
  description: Total number of anomalies detected this cycle.
  constraints:
    - Must equal len(anomalies)
    - Must be >= 0

FIELD_11:
  name:        anomalies
  type:        array of objects
  required:    true
  nullable:    false
  description: >
    Structured anomaly records. Each entry describes one detected deviation
    from the operational baseline. Empty array [] when anomaly_count == 0.
  item_schema:
    id:
      type:        string
      format:      AN-{NN}
      description: Sequential identifier from AN-01; no gaps.
    metric:
      type:        string
      description: The metric name from the source quality or performance report.
    source_report_ref:
      type:        string
      description: report_id of the source report (QR or PR) containing this metric.
    expected_value:
      type:        string
      description: Baseline expected value for this metric this cycle.
    actual_value:
      type:        string
      description: Observed value from the source report.
    deviation_magnitude:
      type:        string
      description: Absolute deviation (actual minus expected, in metric units).
    deviation_pct:
      type:        number
      description: Relative deviation as decimal percentage (e.g. 0.15 = 15%).
    severity:
      type:        enum
      values:      [warning, critical]
      description: >
        warning: deviation exceeds warning threshold but not breach threshold.
        critical: deviation exceeds breach threshold.

FIELD_12:
  name:        escalation_trigger_count
  type:        integer
  required:    true
  nullable:    false
  description: Count of anomalies with severity: critical. Must equal
               len(anomalies filtered by severity == "critical").
  constraints:
    - Must be >= 0
    - When > 0, escalation_required in ESCALATION must be true

## REQUIRED_SECTIONS

SUMMARY:
  heading:     ## SUMMARY
  required:    true
  position:    1
  constraints:
    - Exactly 2–3 sentences
    - Must state anomaly_count and escalation_trigger_count
    - Must state whether baseline_ref is null (bootstrap window)
    - Must state whether escalation was triggered (yes/no)

SCOPE:
  heading:     ## SCOPE
  required:    true
  position:    2
  constraints:
    - Must name the source quality report and performance report by ID
    - Must state the baseline_ref or explicitly note 'no baseline
      (bootstrap window)' when null
    - At least one Covers: and one Excludes: statement

PERIOD:
  heading:     ## PERIOD
  required:    true
  position:    3
  required_fields:
    period_start:              Timestamp — mirrors FIELD_04
    period_end:                Timestamp — mirrors FIELD_05
    prior_report_ref:          Report ID or null — mirrors FIELD_06
    source_quality_report_ref: Report ID — mirrors FIELD_07
    source_performance_report_ref: Report ID — mirrors FIELD_08
    baseline_ref:              Reference or null — mirrors FIELD_09
    context_notes:             >
      Must state 'bootstrap window — no baseline established' when
      baseline_ref is null. Null otherwise.

FINDINGS:
  heading:     ## FINDINGS
  required:    true
  position:    4
  description: >
    One finding per detected anomaly. Findings reference the anomalies
    array entries by AN id. When anomaly_count == 0, FINDINGS must contain
    one explicit 'no anomalies detected' info finding.
  finding_schema:
    id:            string — F prefix + two-digit sequential number
    category:      string — metric name or anomaly category
    finding:       string — quantified deviation fact; present tense
    trend:         enum — [improving, stable, degrading, insufficient-data]
    severity:      enum — [info, warning, critical]
    evidence_refs: array of strings — EV ids
  constraints:
    - F ids sequential from F01 with no gaps
    - Each AN-{NN} entry in anomalies must correspond to at least one finding

EVIDENCE:
  heading:     ## EVIDENCE
  required:    true
  position:    5
  evidence_schema:
    id:              string — EV prefix + two-digit number
    source_artefact: string — QR or PR report_id, or baseline ref
    data_type:       enum — [quality-report, performance-report,
                     baseline-value, audit-event]
    value:           string — extracted value or deviation
    period:          string — time period this evidence covers
  constraints:
    - Minimum 1 entry
    - EV ids sequential from EV01 with no gaps
    - Every finding's evidence_refs must resolve to valid EV ids

ANALYSIS:
  heading:     ## ANALYSIS
  required:    true
  position:    6
  analysis_schema:
    finding_ref:          string — F id
    interpretation:       string — significance, root cause, signal vs noise
    contributing_factors: array of strings — optional
    pattern_ref:          string or null
  constraints:
    - Every finding with severity: warning or critical requires an entry
    - Must explicitly note when baseline_ref is null that baseline
      comparison could not be performed

RECOMMENDATIONS:
  heading:     ## RECOMMENDATIONS
  required:    true
  position:    7
  recommendation_schema:
    id:              string — R prefix + two-digit number
    finding_refs:    array of strings — F ids
    recommendation:  string — present tense, actionable
    directed_to:     string — agent:[slug] or sysadmin:dlm-sysadmin
    priority:        enum — [low, medium, high, immediate]
    expected_impact: string
  constraints:
    - May be empty [] if no actionable findings
    - If escalation_required == true, at least one recommendation with
      priority: immediate

ESCALATION:
  heading:     ## ESCALATION
  required:    true
  position:    8
  required_fields:
    escalation_required: >
      Boolean. True if escalation_trigger_count > 0 (any critical anomaly).
    escalation_basis:    >
      Array of AN ids for critical anomalies. Empty [] when
      escalation_required == false.
    escalation_target:   >
      Must be sysadmin:dlm-sysadmin when escalation_required == true.
      Null otherwise.
    notification_sent:   Boolean. True once notification dispatched.

CHANGE_LOG:
  heading:     ## CHANGE_LOG
  required:    true
  position:    last
  format:      "- v[version] | [YYYY-MM-DD] | [agent_id] | [change summary]"

## OPTIONAL_SECTIONS

NOTES:
  heading:      ## NOTES
  description:  Non-normative commentary on detection methodology or
                bootstrap-phase constraints.
  when_to_include: >
    Bootstrap window disclosures, detection algorithm notes, or
    known data-quality issues with source reports.
  position:     After CHANGE_LOG

## VALIDATION_RULES

V01:
  rule:     report_id is non-null, matches AR-{YYYY}-{NNNN}, and is unique
            for the year.
  test:     >
    report_id != null AND
    report_id matches regex ^AR-\d{4}-\d{4}$ AND
    no other anomaly report for the same year shares this id.
  checked_by: CI Cycle Orchestrator; Audit Trail Agent

V02:
  rule:     generated_by must be agent:anomaly-detector.
  test:     generated_by == "agent:anomaly-detector"
  checked_by: CI Cycle Orchestrator

V03:
  rule:     period_end must be >= period_start.
  test:     period_end >= period_start
  checked_by: CI Cycle Orchestrator

V04:
  rule:     anomaly_count must equal the length of the anomalies array.
  test:     anomaly_count == len(anomalies)
  checked_by: Agent Format Validator; Anomaly Detector

V05:
  rule:     All AN-{NN} ids in anomalies must be sequential from AN-01
            with no gaps.
  test:     >
    for i, A in enumerate(anomalies):
      A.id == "AN-" + zero_pad(i+1, 2)
  checked_by: Agent Format Validator

V06:
  rule:     escalation_trigger_count must equal the count of anomalies
            with severity: critical.
  test:     >
    escalation_trigger_count ==
      len([A for A in anomalies where A.severity == "critical"])
  checked_by: Agent Format Validator; Anomaly Detector

V07:
  rule:     escalation_required in ESCALATION must be true when
            escalation_trigger_count > 0.
  test:     >
    IF escalation_trigger_count > 0
    THEN escalation_required == true
  checked_by: CI Cycle Orchestrator; Audit Trail Agent

V08:
  rule:     baseline_ref must be non-null unless the artefact is produced
            within the 90-day bootstrap window. When null, context_notes
            in PERIOD must state 'bootstrap window — no baseline established'.
  test:     >
    IF (current_date - system_activation_date) > 90 days
    THEN baseline_ref != null
    ELSE if baseline_ref == null
      THEN PERIOD.context_notes contains 'bootstrap window'
  checked_by: Anomaly Detector; Agent Format Validator

V09:
  rule:     All nine required sections must be present and appear in the
            prescribed order.
  test:     >
    Sections SUMMARY, SCOPE, PERIOD, FINDINGS, EVIDENCE, ANALYSIS,
    RECOMMENDATIONS, ESCALATION, CHANGE_LOG all present and in sequence.
  checked_by: Agent Format Validator; Template Validator
