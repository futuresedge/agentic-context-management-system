## SUMMARY
This template defines the schema and structure of a quality report artefact
produced by the Quality Monitor agent at the end of each CI cycle. It
operationalises the four mandatory measurement categories from
DLMS-2026-0043 R11 — verification conformance, DoD conformance, hard-fail
breaches, and time-threshold exceedance — into a machine-readable, auditable
artefact consumed by the Anomaly Detector and CI Perspective Synthesizer.

## TARGET_DOC_TYPE
quality-report

## SCHEMA

### Fields

FIELD_01:
  name:        report_id
  type:        string
  required:    true
  nullable:    false
  format:      QR-{YYYY}-{NNNN}
  example:     QR-2026-0001
  description: Unique identifier for this quality report. NNNN is a
               zero-padded sequential integer scoped to the year.
               Immutable after assignment.

FIELD_02:
  name:        cycle_id
  type:        string
  required:    true
  nullable:    false
  format:      CYCLE-{YYYY}-{NNNN}
  description: The CI cycle this report belongs to.

FIELD_03:
  name:        generated_by
  type:        string
  required:    true
  nullable:    false
  format:      agent:[agent-role-slug]
  constraints:
    - Must be agent:quality-monitor
    - Must differ from verified_by (Verification Independence Policy
      DLMS-2026-0014)

FIELD_04:
  name:        period_start
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset
  description: Start of the reporting period (inclusive).

FIELD_05:
  name:        period_end
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset
  constraints:
    - period_end must be >= period_start

FIELD_06:
  name:        prior_report_ref
  type:        string or null
  required:    true
  nullable:    true
  description: >
    The report_id of the most recent prior quality report, if one exists.
    Null for the first report. Enables trend comparison across CI cycles.

FIELD_07:
  name:        documents_assessed
  type:        integer
  required:    true
  nullable:    false
  description: Count of documents assessed during this CI cycle period.
  constraints:
    - Must be >= 0

FIELD_08:
  name:        quality_metrics
  type:        object
  required:    true
  nullable:    false
  description: >
    Structured quality measurement data. Contains exactly four sub-fields
    as mandated by DLMS-2026-0043 R11. All sub-fields are required and
    must be non-null.
  sub_fields:
    verification_conformance:
      type:        object
      description: Verification record conformance rates by lens and doc_type.
      structure:
        by_lens:
          type:    array of objects
          fields:  [{lens, conformant, total, rate}]
          note:    rate = conformant / total expressed as decimal 0–1
        by_doc_type:
          type:    array of objects
          fields:  [{doc_type, conformant, total, rate}]
    dod_conformance:
      type:        object
      description: DoD record conformance rates by stage.
      structure:
        by_stage:
          type:    array of objects
          fields:  [{stage, conformant, total, rate}]
    hard_fail_breaches:
      type:        array of objects
      fields:      [{criterion_ref, count}]
      note:        criterion_ref references criterion identifiers from
                   DLMS-2026-0043 R03–R08 (e.g. Q01, Q02…).
                   Empty array [] when no breaches occurred.
    time_threshold_exceeded:
      type:        array of objects
      fields:      [{threshold_type, stage, count}]
      note:        threshold_type matches the threshold identifier in
                   DLMS-2026-0043 R09. Empty [] when no exceedances.

FIELD_09:
  name:        overall_quality_status
  type:        enum
  required:    true
  nullable:    false
  values:      [passing, warning, critical]
  derivation_rules:
    critical: >
      Any entry in hard_fail_breaches has count > 0.
    warning: >
      No hard_fail_breaches AND any conformance rate is below the warning
      threshold defined in DLMS-2026-0043 R11.
    passing: >
      All four metric categories within target thresholds.

## REQUIRED_SECTIONS

SUMMARY:
  heading:     ## SUMMARY
  required:    true
  position:    1
  constraints:
    - Exactly 2–3 sentences
    - Must state overall_quality_status value expressly
    - Must state whether escalation was triggered (yes/no)
    - Must name the cycle_id and documents_assessed count

SCOPE:
  heading:     ## SCOPE
  required:    true
  position:    2
  constraints:
    - At least one Covers: statement listing source artefact types
      consumed (verification records, DoD records)
    - At least one Excludes: statement
    - Must list all source artefact paths consumed

PERIOD:
  heading:     ## PERIOD
  required:    true
  position:    3
  required_fields:
    period_start:       Timestamp — must mirror FIELD_04
    period_end:         Timestamp — must mirror FIELD_05
    documents_assessed: Integer — must mirror FIELD_07
    prior_report_ref:   Report ID or null — must mirror FIELD_06
    context_notes:      >
      Operational events during this period that may affect metric
      interpretation (e.g. bootstrap phase, known data gaps). Null if none.

FINDINGS:
  heading:     ## FINDINGS
  required:    true
  position:    4
  description: >
    One finding per quality_metrics category or derived observation.
    Findings are factual only — interpretation belongs in ANALYSIS.
  finding_schema:
    id:            string — F prefix + two-digit sequential number (F01, F02…)
    category:      enum — [verification-conformance, dod-conformance,
                   hard-fail-breach, time-threshold]
    finding:       string — Observed fact. Quantified. Present tense.
    trend:         enum — [improving, stable, degrading, insufficient-data]
    severity:      enum — [info, warning, critical]
    evidence_refs: array of strings — EV ids supporting this finding
  constraints:
    - Minimum 1 finding
    - F ids sequential from F01 with no gaps
    - A finding with severity: critical triggers escalation_required: true
      in the ESCALATION section

EVIDENCE:
  heading:     ## EVIDENCE
  required:    true
  position:    5
  evidence_schema:
    id:              string — EV prefix + two-digit sequential number
    source_artefact: string — file path or artefact ID of source record
    data_type:       enum — [verification-record, dod-record,
                     audit-event-count]
    value:           string — extracted metric or count value
    period:          string — time period this evidence covers
  constraints:
    - Minimum 1 entry
    - EV ids sequential from EV01 with no gaps
    - Every finding's evidence_refs must resolve to valid EV ids in this section

ANALYSIS:
  heading:     ## ANALYSIS
  required:    true
  position:    6
  analysis_schema:
    finding_ref:          string — F id being analysed
    interpretation:       string — what this finding means; root cause if known
    contributing_factors: array of strings — optional; known or suspected causes
    pattern_ref:          string or null — KB doc_id if finding matches a
                          known pattern; null otherwise
  constraints:
    - Every finding with severity: warning or critical must have a
      corresponding analysis entry
    - Info findings may be listed without analysis if self-evident

RECOMMENDATIONS:
  heading:     ## RECOMMENDATIONS
  required:    true
  position:    7
  recommendation_schema:
    id:              string — R prefix + two-digit sequential number
    finding_refs:    array of strings — F ids motivating this recommendation
    recommendation:  string — present tense, actionable, specific
    directed_to:     string — agent:[slug] or sysadmin:dlm-sysadmin
    priority:        enum — [low, medium, high, immediate]
    expected_impact: string — quantified improvement expected
  constraints:
    - May be empty [] if no actionable findings arise
    - If escalation_required == true, at least one recommendation must
      have priority: immediate

ESCALATION:
  heading:     ## ESCALATION
  required:    true
  position:    8
  required_fields:
    escalation_required: >
      Boolean. True if overall_quality_status == critical OR any finding
      has severity: critical.
    escalation_basis:    >
      Array of F ids triggering escalation. Empty [] when
      escalation_required == false.
    escalation_target:   >
      The agent or role to notify. Null when escalation_required == false.
      Must be sysadmin:dlm-sysadmin for critical escalations.
    notification_sent:   >
      Boolean. True once notification has been dispatched.

CHANGE_LOG:
  heading:     ## CHANGE_LOG
  required:    true
  position:    last
  format:      "- v[version] | [YYYY-MM-DD] | [agent_id] | [change summary]"
  constraints:
    - Newest version first; one entry per version; immutable once written

## OPTIONAL_SECTIONS

NOTES:
  heading:      ## NOTES
  description:  Non-normative commentary or operational caveats.
  when_to_include: >
    Bootstrap-phase interpretation notes, data gap disclosures, or
    unusual cycle circumstances not captured in ANALYSIS.
  position:     After CHANGE_LOG

## VALIDATION_RULES

V01:
  rule:     report_id is non-null, matches QR-{YYYY}-{NNNN}, and is unique
            for the year.
  test:     >
    report_id != null AND
    report_id matches regex ^QR-\d{4}-\d{4}$ AND
    no other quality report artefact for the same year shares this id.
  checked_by: CI Cycle Orchestrator; Audit Trail Agent

V02:
  rule:     generated_by must be agent:quality-monitor and must differ from
            the agent that performs verification (Verification Independence
            Policy DLMS-2026-0014).
  test:     >
    generated_by == "agent:quality-monitor" AND
    generated_by != verified_by (if verified_by is set)
  checked_by: CI Cycle Orchestrator; Agent Format Validator

V03:
  rule:     period_end must be >= period_start.
  test:     period_end >= period_start
  checked_by: CI Cycle Orchestrator

V04:
  rule:     quality_metrics must contain exactly the four required sub-fields:
            verification_conformance, dod_conformance, hard_fail_breaches,
            and time_threshold_exceeded. No sub-field may be absent or null.
  test:     >
    quality_metrics.verification_conformance != null AND
    quality_metrics.dod_conformance != null AND
    quality_metrics.hard_fail_breaches != null AND
    quality_metrics.time_threshold_exceeded != null
  checked_by: Agent Format Validator; Quality Monitor

V05:
  rule:     overall_quality_status must be correctly derived from
            quality_metrics per the FIELD_09 derivation rules.
  test:     >
    IF any quality_metrics.hard_fail_breaches[*].count > 0
      THEN overall_quality_status == "critical"
    ELSE IF any conformance rate in verification_conformance.by_lens or
            dod_conformance.by_stage is below the DLMS-2026-0043 R11
            warning threshold
      THEN overall_quality_status == "warning"
    ELSE overall_quality_status == "passing"
  checked_by: Quality Monitor; Agent Format Validator

V06:
  rule:     Every finding with severity: warning or critical must have a
            corresponding entry in the ANALYSIS section.
  test:     >
    For all F in FINDINGS where F.severity IN ["warning", "critical"]:
      EXISTS A in ANALYSIS where A.finding_ref == F.id
  checked_by: CI Perspective Synthesizer; Agent Format Validator

V07:
  rule:     escalation_required must be true when overall_quality_status is
            critical or any finding has severity: critical.
  test:     >
    IF (overall_quality_status == "critical" OR
        any FINDINGS[*].severity == "critical")
    THEN escalation_required == true
  checked_by: CI Cycle Orchestrator; Audit Trail Agent

V08:
  rule:     All nine required sections must be present and appear in the
            prescribed order.
  test:     >
    Sections SUMMARY, SCOPE, PERIOD, FINDINGS, EVIDENCE, ANALYSIS,
    RECOMMENDATIONS, ESCALATION, CHANGE_LOG all present and in sequence.
  checked_by: Agent Format Validator; Template Validator
