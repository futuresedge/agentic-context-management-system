## SUMMARY
This template defines the schema and structure of a CI synthesis artefact
produced by the CI Perspective Synthesizer once all five monitor reports
for a CI cycle are available. It integrates quality, performance, anomaly,
bottleneck, and agent behaviour findings into a unified multi-perspective
view — without containing final recommendations, which are deferred to the
Recommendation Generator.

## TARGET_DOC_TYPE
ci-synthesis

## SCHEMA

### Fields

FIELD_01:
  name:        report_id
  type:        string
  required:    true
  nullable:    false
  format:      CIS-{YYYY}-{NNNN}
  example:     CIS-2026-0001
  description: Unique identifier for this CI synthesis artefact. NNNN is a
               zero-padded sequential integer scoped to the year.
               Immutable after assignment.

FIELD_02:
  name:        cycle_id
  type:        string
  required:    true
  nullable:    false
  format:      CYCLE-{YYYY}-{NNNN}

FIELD_03:
  name:        generated_by
  type:        string
  required:    true
  nullable:    false
  format:      agent:[agent-role-slug]
  constraints:
    - Must be agent:ci-perspective-synthesizer
    - Must differ from verified_by (Verification Independence Policy
      DLMS-2026-0014)

FIELD_04:
  name:        period_start
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset

FIELD_05:
  name:        period_end
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset
  constraints:
    - period_end must be >= period_start

FIELD_06:
  name:        prior_report_ref
  type:        string or null
  required:    true
  nullable:    true
  description: >
    The report_id of the most recent prior CI synthesis artefact. Null for
    the first synthesis. Enables cross-cycle theme comparison.

FIELD_07:
  name:        source_reports
  type:        object
  required:    true
  nullable:    false
  description: >
    References to all five monitor report artefacts for this CI cycle.
    All five fields are required and must be non-null. The CI Perspective
    Synthesizer must not produce a synthesis artefact until all five
    source reports are available.
  structure:
    quality_report_ref:
      type:        string
      description: QR-{YYYY}-{NNNN} identifier of the quality report.
    performance_report_ref:
      type:        string
      description: PR-{YYYY}-{NNNN} identifier of the performance report.
    anomaly_report_ref:
      type:        string
      description: AR-{YYYY}-{NNNN} identifier of the anomaly report.
    bottleneck_report_ref:
      type:        string
      description: BNR-{YYYY}-{NNNN} identifier of the bottleneck report.
    behaviour_report_ref:
      type:        string
      description: ABR-{YYYY}-{NNNN} identifier of the agent behaviour report.

FIELD_08:
  name:        escalation_carried_forward
  type:        boolean
  required:    true
  nullable:    false
  description: >
    True if at least one of the five source reports had
    escalation_required: true. The CI Perspective Synthesizer must
    propagate the escalation signal upward even if the synthesis artefact
    itself does not add new critical findings.

FIELD_09:
  name:        theme_count
  type:        integer
  required:    true
  nullable:    false
  description: >
    Total number of cross-cutting themes identified in FINDINGS.
    Must equal len(FINDINGS) entries where category: cross-cutting-theme.
  constraints:
    - Must be >= 0

## REQUIRED_SECTIONS

SUMMARY:
  heading:     ## SUMMARY
  required:    true
  position:    1
  constraints:
    - Exactly 2–3 sentences
    - Must state theme_count and escalation_carried_forward value
    - Must name the five source reports by ID
    - Must state whether any new findings arise beyond escalation propagation

SCOPE:
  heading:     ## SCOPE
  required:    true
  position:    2
  constraints:
    - Must name all five source reports by ID and confirm their
      availability before synthesis was triggered
    - At least one Covers: statement, one Excludes: statement
    - Must explicitly state: "This artefact does not contain final
      action recommendations — those are produced by the Recommendation
      Generator from this artefact"

PERIOD:
  heading:     ## PERIOD
  required:    true
  position:    3
  required_fields:
    period_start:             Timestamp — mirrors FIELD_04
    period_end:               Timestamp — mirrors FIELD_05
    quality_report_ref:       mirrors source_reports.quality_report_ref
    performance_report_ref:   mirrors source_reports.performance_report_ref
    anomaly_report_ref:       mirrors source_reports.anomaly_report_ref
    bottleneck_report_ref:    mirrors source_reports.bottleneck_report_ref
    behaviour_report_ref:     mirrors source_reports.behaviour_report_ref
    prior_report_ref:         mirrors FIELD_06
    context_notes:            Null if none

FINDINGS:
  heading:     ## FINDINGS
  required:    true
  position:    4
  description: >
    Cross-cutting themes that span two or more of the five source reports.
    Each finding describes a systemic pattern or correlated signal that
    emerges from synthesising across multiple dimensions. Findings that
    appear in only one source report are not repeated here — they remain
    in the source report only and are referenced by source_report_ref.
    The RECOMMENDATIONS section in this artefact contains routing
    directives only, not action recommendations.
  finding_schema:
    id:              string — F prefix + two-digit sequential number
    category:        enum — [cross-cutting-theme, escalation-propagation,
                     correlated-signal]
    finding:         string — present tense, factual; identifies which
                     source reports contribute to this theme
    source_reports:  array of strings — subset of source report IDs
                     contributing to this finding (minimum 2 for
                     cross-cutting-theme; may be 1 for escalation-propagation)
    trend:           enum — [improving, stable, degrading, insufficient-data]
    severity:        enum — [info, warning, critical]
    evidence_refs:   array of strings — EV ids
  constraints:
    - Minimum 1 finding (at minimum an escalation-propagation finding if
      escalation_carried_forward is true but no themes were identified)
    - F ids sequential from F01 with no gaps
    - Cross-cutting-theme findings must cite at least 2 source reports
    - theme_count equals count of findings with category: cross-cutting-theme

EVIDENCE:
  heading:     ## EVIDENCE
  required:    true
  position:    5
  evidence_schema:
    id:              string — EV prefix + two-digit number
    source_artefact: string — one of the five source report IDs
    data_type:       enum — [quality-report, performance-report,
                     anomaly-report, bottleneck-report,
                     agent-behaviour-report]
    value:           string — specific finding ID or metric from source report
    period:          string — cycle period reference
  constraints:
    - Minimum 1 entry
    - EV ids sequential from EV01 with no gaps
    - Evidence must trace to specific findings in the named source reports

ANALYSIS:
  heading:     ## ANALYSIS
  required:    true
  position:    6
  analysis_schema:
    finding_ref:          string — F id
    interpretation:       string — systemic significance of the cross-cutting
                          theme; what the convergence of signals means
    contributing_factors: array of strings — optional
    pattern_ref:          string or null
  constraints:
    - Every finding with severity: warning or critical requires an entry
    - Analysis must explicitly note when a finding represents a new theme
      not present in prior_report_ref vs a recurring pattern

RECOMMENDATIONS:
  heading:     ## RECOMMENDATIONS
  required:    true
  position:    7
  description: >
    Routing directives only. This section does not contain action
    recommendations — those are produced by the Recommendation Generator
    (TMPL-recommendations-001) from this artefact. Entries here direct
    this artefact to appropriate consumers.
  recommendation_schema:
    id:              string — R prefix + two-digit number
    finding_refs:    array of strings — F ids
    recommendation:  string — routing directive (e.g. "Route to Recommendation
                     Generator for action recommendation generation")
    directed_to:     string — agent:[slug] or sysadmin:dlm-sysadmin
    priority:        enum — [low, medium, high, immediate]
    expected_impact: string — routing purpose
  constraints:
    - Minimum 1 entry routing the synthesis artefact to the
      Recommendation Generator
    - When escalation_carried_forward is true, an additional entry with
      priority: immediate must route the escalation to
      sysadmin:dlm-sysadmin directly

ESCALATION:
  heading:     ## ESCALATION
  required:    true
  position:    8
  required_fields:
    escalation_required: >
      Boolean. True when escalation_carried_forward == true OR any
      finding in this artefact has severity: critical.
    escalation_basis:    >
      Array identifying the source reports that carried escalation
      and/or F ids for new critical findings. Empty [] when
      escalation_required == false.
    escalation_target:   sysadmin:dlm-sysadmin or null.
    notification_sent:   Boolean. True once notification dispatched.

CHANGE_LOG:
  heading:     ## CHANGE_LOG
  required:    true
  position:    last
  format:      "- v[version] | [YYYY-MM-DD] | [agent_id] | [change summary]"

## OPTIONAL_SECTIONS

NOTES:
  heading:      ## NOTES
  description:  Non-normative commentary on synthesis methodology or
                theme detection approach.
  when_to_include: >
    First-cycle notes on theme-detection calibration or methodology
    changes between cycles.
  position:     After CHANGE_LOG

## VALIDATION_RULES

V01:
  rule:     report_id is non-null, matches CIS-{YYYY}-{NNNN}, and is unique
            for the year.
  test:     >
    report_id != null AND
    report_id matches regex ^CIS-\d{4}-\d{4}$ AND
    no other CI synthesis artefact for the same year shares this id.
  checked_by: CI Cycle Orchestrator; Audit Trail Agent

V02:
  rule:     generated_by must be agent:ci-perspective-synthesizer.
  test:     generated_by == "agent:ci-perspective-synthesizer"
  checked_by: CI Cycle Orchestrator

V03:
  rule:     period_end must be >= period_start.
  test:     period_end >= period_start
  checked_by: CI Cycle Orchestrator

V04:
  rule:     All five fields in source_reports must be non-null and each
            must match the corresponding report type ID format.
  test:     >
    source_reports.quality_report_ref matches ^QR-\d{4}-\d{4}$ AND
    source_reports.performance_report_ref matches ^PR-\d{4}-\d{4}$ AND
    source_reports.anomaly_report_ref matches ^AR-\d{4}-\d{4}$ AND
    source_reports.bottleneck_report_ref matches ^BNR-\d{4}-\d{4}$ AND
    source_reports.behaviour_report_ref matches ^ABR-\d{4}-\d{4}$
  checked_by: Agent Format Validator; CI Cycle Orchestrator

V05:
  rule:     escalation_carried_forward must be true if any of the five
            source reports had escalation_required: true.
  test:     >
    IF any source_report.escalation_required == true
    THEN escalation_carried_forward == true
  checked_by: CI Perspective Synthesizer; Agent Format Validator

V06:
  rule:     theme_count must equal the count of FINDINGS entries with
            category: cross-cutting-theme.
  test:     >
    theme_count == len([F for F in FINDINGS
                        where F.category == "cross-cutting-theme"])
  checked_by: Agent Format Validator

V07:
  rule:     Cross-cutting-theme findings must cite at least two source
            reports in their source_reports array.
  test:     >
    For all F in FINDINGS where F.category == "cross-cutting-theme":
      len(F.source_reports) >= 2
  checked_by: Agent Format Validator; CI Perspective Synthesizer

V08:
  rule:     RECOMMENDATIONS must contain at minimum one routing entry
            directed to the Recommendation Generator, and when
            escalation_carried_forward is true, one additional entry
            with priority: immediate directed to sysadmin:dlm-sysadmin.
  test:     >
    EXISTS R in RECOMMENDATIONS where R.directed_to == "agent:recommendation-generator"
    AND IF escalation_carried_forward == true THEN
      EXISTS R in RECOMMENDATIONS where
        R.directed_to == "sysadmin:dlm-sysadmin" AND
        R.priority == "immediate"
  checked_by: CI Cycle Orchestrator; Agent Format Validator

V09:
  rule:     All nine required sections must be present and appear in the
            prescribed order.
  test:     >
    Sections SUMMARY, SCOPE, PERIOD, FINDINGS, EVIDENCE, ANALYSIS,
    RECOMMENDATIONS, ESCALATION, CHANGE_LOG all present and in sequence.
  checked_by: Agent Format Validator; Template Validator
