## SUMMARY
This template defines the schema and structure of an improvement
recommendations artefact produced by the Recommendation Generator agent
after receiving a CI synthesis artefact. It translates cross-cutting themes
and escalated signals from the synthesis into actionable, prioritised
directives routed to appropriate agents and the SysAdmin Briefing Agent.

## TARGET_DOC_TYPE
improvement-recommendations

## SCHEMA

### Fields

FIELD_01:
  name:        report_id
  type:        string
  required:    true
  nullable:    false
  format:      IMP-{YYYY}-{NNNN}
  example:     IMP-2026-0001
  description: Unique identifier for this recommendations artefact. NNNN is
               a zero-padded sequential integer scoped to the year.
               Immutable after assignment.

FIELD_02:
  name:        cycle_id
  type:        string
  required:    true
  nullable:    false
  format:      CYCLE-{YYYY}-{NNNN}

FIELD_03:
  name:        generated_by
  type:        string
  required:    true
  nullable:    false
  format:      agent:[agent-role-slug]
  constraints:
    - Must be agent:recommendation-generator
    - Must differ from verified_by (Verification Independence Policy
      DLMS-2026-0014)

FIELD_04:
  name:        period_start
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset

FIELD_05:
  name:        period_end
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset
  constraints:
    - period_end must be >= period_start

FIELD_06:
  name:        prior_report_ref
  type:        string or null
  required:    true
  nullable:    true
  description: >
    The report_id of the most recent prior improvement recommendations
    artefact. Null for the first artefact. Used for trend comparison
    and tracking recommendation recurrence.

FIELD_07:
  name:        source_synthesis_ref
  type:        string
  required:    true
  nullable:    false
  format:      CIS-{YYYY}-{NNNN}
  description: >
    The report_id of the CI synthesis artefact that this recommendations
    artefact is derived from. The Recommendation Generator must not produce
    this artefact without a valid CI synthesis artefact as input.

FIELD_08:
  name:        recommendation_count
  type:        integer
  required:    true
  nullable:    false
  description: Total number of recommendations in the RECOMMENDATIONS section.
  constraints:
    - Must be >= 0
    - Must equal len(RECOMMENDATIONS entries)

FIELD_09:
  name:        critical_recommendation_count
  type:        integer
  required:    true
  nullable:    false
  description: Count of recommendations with priority: immediate.
  constraints:
    - Must be >= 0
    - Must equal len([R in RECOMMENDATIONS where R.priority == "immediate"])

FIELD_10:
  name:        immediate_action_required
  type:        boolean
  required:    true
  nullable:    false
  description: >
    True when critical_recommendation_count > 0. Signals to the SysAdmin
    Briefing Agent that at least one recommendation demands immediate action.

## REQUIRED_SECTIONS

SUMMARY:
  heading:     ## SUMMARY
  required:    true
  position:    1
  constraints:
    - Exactly 2–3 sentences
    - Must state recommendation_count and critical_recommendation_count
    - Must state immediate_action_required value
    - Must reference the source_synthesis_ref

SCOPE:
  heading:     ## SCOPE
  required:    true
  position:    2
  constraints:
    - Must name the CI synthesis artefact (source_synthesis_ref)
    - Must state that recommendations are derived from cross-cutting themes
      in the synthesis, not restated from individual source reports
    - At least one Covers: and one Excludes: statement

PERIOD:
  heading:     ## PERIOD
  required:    true
  position:    3
  required_fields:
    period_start:           Timestamp — mirrors FIELD_04
    period_end:             Timestamp — mirrors FIELD_05
    source_synthesis_ref:   CIS ID — mirrors FIELD_07
    prior_report_ref:       IMP ID or null — mirrors FIELD_06
    context_notes:          Null if none

FINDINGS:
  heading:     ## FINDINGS
  required:    true
  position:    4
  description: >
    Findings surfaced from the CI synthesis artefact that drive
    specific recommendations. Not a repeat of synthesis FINDINGS;
    rather, the subset of synthesised themes that have actionable
    response paths.
  finding_schema:
    id:              string — F prefix + two-digit sequential number
    synthesis_ref:   string — F-id from source synthesis FINDINGS
    finding:         string — restatement of synthesised theme with
                     recommendation context
    recurrence:      enum — [first-occurrence, recurring, persistent]
    severity:        enum — [info, warning, critical]
    evidence_refs:   array of strings — EV ids
  constraints:
    - At least 1 finding if recommendation_count > 0
    - F ids sequential from F01 with no gaps
    - Each finding maps to exactly one recommendation (many findings
      may map to the same recommendation via finding_refs)

EVIDENCE:
  heading:     ## EVIDENCE
  required:    true
  position:    5
  evidence_schema:
    id:              string — EV prefix + two-digit number
    source_artefact: string — CIS report ID and/or underlying monitor
                     report ID
    data_type:       enum — [ci-synthesis, quality-report, performance-report,
                     anomaly-report, bottleneck-report,
                     agent-behaviour-report]
    value:           string — relevant finding ID or metric
    period:          string — cycle period reference
  constraints:
    - Minimum 1 entry when recommendation_count > 0; otherwise []
    - EV ids sequential from EV01 with no gaps

ANALYSIS:
  heading:     ## ANALYSIS
  required:    true
  position:    6
  analysis_schema:
    finding_ref:          string — F id
    action_rationale:     string — why this finding requires the recommended
                          action
    recurrence_context:   string or null — context if marked recurring
                          or persistent; null for first-occurrence
    policy_refs:          array of strings — DLMS policy doc_ids that
                          mandate or motivate the recommendation
  constraints:
    - Every finding with severity: warning or critical requires an entry
    - When recurrence: persistent, analysis must identify the
      specific cycle(s) in which the finding previously appeared

RECOMMENDATIONS:
  heading:     ## RECOMMENDATIONS
  required:    true
  position:    7
  description: >
    Actionable, prioritised directives in response to findings. These
    are the final action recommendations for the CI cycle.
  recommendation_schema:
    id:              string — R prefix + two-digit sequential number
    finding_refs:    array of strings — F ids supported by this recommendation
    recommendation:  string — present tense, specific, actionable
    directed_to:     string — agent:[slug] or sysadmin:dlm-sysadmin
    priority:        enum — [low, medium, high, immediate]
    expected_impact: string — quantified where possible
    policy_refs:     array of strings — DLMS doc_ids requiring this action;
                     empty [] if no policy mandate
  constraints:
    - May be empty [] if no actionable findings
    - If immediate_action_required == true, at least one recommendation
      with priority: immediate must be present
    - R ids sequential from R01 with no gaps

ESCALATION:
  heading:     ## ESCALATION
  required:    true
  position:    8
  required_fields:
    escalation_required: >
      Boolean. True when any recommendation has priority: immediate OR
      when the source CI synthesis carried escalation_carried_forward: true.
    escalation_basis:    >
      Array of R ids with priority: immediate and/or note of synthesised
      escalation carry-forward. Empty [] when escalation_required == false.
    escalation_target:   sysadmin:dlm-sysadmin or null.
    notification_sent:   Boolean.

CHANGE_LOG:
  heading:     ## CHANGE_LOG
  required:    true
  position:    last
  format:      "- v[version] | [YYYY-MM-DD] | [agent_id] | [change summary]"

## OPTIONAL_SECTIONS

NOTES:
  heading:      ## NOTES
  description:  Commentary on recommendation prioritisation methodology or
                cross-cycle recurrence analysis.
  when_to_include: When methodology choices need explicit documentation.
  position:     After CHANGE_LOG

## VALIDATION_RULES

V01:
  rule:     report_id is non-null, matches IMP-{YYYY}-{NNNN}, and is unique
            for the year.
  test:     >
    report_id != null AND
    report_id matches regex ^IMP-\d{4}-\d{4}$ AND
    no other improvement-recommendations artefact for the same year
    shares this id.
  checked_by: CI Cycle Orchestrator; Audit Trail Agent

V02:
  rule:     generated_by must be agent:recommendation-generator.
  test:     generated_by == "agent:recommendation-generator"
  checked_by: CI Cycle Orchestrator

V03:
  rule:     source_synthesis_ref must be non-null and match the
            CIS-{YYYY}-{NNNN} format.
  test:     >
    source_synthesis_ref != null AND
    source_synthesis_ref matches regex ^CIS-\d{4}-\d{4}$
  checked_by: Agent Format Validator; CI Cycle Orchestrator

V04:
  rule:     recommendation_count must equal the actual count of
            entries in the RECOMMENDATIONS section.
  test:     recommendation_count == len(RECOMMENDATIONS entries)
  checked_by: Agent Format Validator

V05:
  rule:     critical_recommendation_count must equal the count of
            RECOMMENDATIONS entries with priority: immediate.
  test:     >
    critical_recommendation_count ==
      len([R for R in RECOMMENDATIONS where R.priority == "immediate"])
  checked_by: Agent Format Validator

V06:
  rule:     immediate_action_required must be true when
            critical_recommendation_count > 0.
  test:     >
    IF critical_recommendation_count > 0
    THEN immediate_action_required == true
  checked_by: Agent Format Validator; Recommendation Generator

V07:
  rule:     Every finding with severity: warning or critical must have
            a corresponding entry in ANALYSIS.
  test:     >
    For all F in FINDINGS where F.severity IN ["warning", "critical"]:
      EXISTS A in ANALYSIS where A.finding_ref == F.id
  checked_by: Agent Format Validator; CI Cycle Orchestrator

V08:
  rule:     All nine required sections must be present and appear in
            the prescribed order.
  test:     >
    Sections SUMMARY, SCOPE, PERIOD, FINDINGS, EVIDENCE, ANALYSIS,
    RECOMMENDATIONS, ESCALATION, CHANGE_LOG all present and in sequence.
  checked_by: Agent Format Validator; Template Validator
