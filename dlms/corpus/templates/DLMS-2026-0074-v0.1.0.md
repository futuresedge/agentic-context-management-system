## SUMMARY
This template defines the schema and structure of a change directive
artefact produced by the DLM SysAdmin Agent, recording the DLM SysAdmin's
formal decisions against the action items presented in the SysAdmin briefing.
It is a decision record, not a monitoring or analysis report; it uses the
DECISION_RECORD section structure rather than the report body structure.

## TARGET_DOC_TYPE
change-directive

## SCHEMA

### Fields

FIELD_01:
  name:        directive_id
  type:        string
  required:    true
  nullable:    false
  format:      CD-{YYYY}-{NNNN}
  example:     CD-2026-0001
  description: Unique identifier for this change directive. NNNN is a
               zero-padded sequential integer scoped to the year.
               Immutable after assignment.

FIELD_02:
  name:        cycle_id
  type:        string
  required:    true
  nullable:    false
  format:      CYCLE-{YYYY}-{NNNN}

FIELD_03:
  name:        issued_by
  type:        string
  required:    true
  nullable:    false
  description: >
    The human role or agent recording the SysAdmin decisions.
  constraints:
    - Must be sysadmin:dlm-sysadmin or agent:dlm-sysadmin-agent
    - Must differ from verified_by (Verification Independence Policy
      DLMS-2026-0014)
  note:         >
    When the DLM SysAdmin makes decisions via a supporting agent interface,
    issued_by captures the human identity. When an agent records decisions
    on behalf of the SysAdmin, agent:dlm-sysadmin-agent is acceptable.

FIELD_04:
  name:        issued_at
  type:        datetime
  required:    true
  nullable:    false
  format:      ISO 8601 with timezone offset
  description: >
    The timestamp at which the DLM SysAdmin issued this change directive.
    Not a period; this is a point-in-time decision record.
  note:         >
    Change directives use issued_at instead of period_start/period_end.
    The CI cycle period is captured in cycle_id and the briefing_ref.

FIELD_05:
  name:        briefing_ref
  type:        string
  required:    true
  nullable:    false
  format:      SBR-{YYYY}-{NNNN}
  description: >
    The SysAdmin briefing artefact that presented the action items on
    which this directive records decisions. Must be non-null.

FIELD_06:
  name:        directive_count
  type:        integer
  required:    true
  nullable:    false
  description: Total number of decision entries in the DIRECTIVES section.
  constraints:
    - Must be >= 1
    - Must equal accepted_count + deferred_count + rejected_count

FIELD_07:
  name:        accepted_count
  type:        integer
  required:    true
  nullable:    false
  description: Count of directives with decision: accepted.
  constraints:
    - Must be >= 0

FIELD_08:
  name:        deferred_count
  type:        integer
  required:    true
  nullable:    false
  description: Count of directives with decision: deferred.
  constraints:
    - Must be >= 0

FIELD_09:
  name:        rejected_count
  type:        integer
  required:    true
  nullable:    false
  description: Count of directives with decision: rejected.
  constraints:
    - Must be >= 0

FIELD_10:
  name:        directives
  type:        array of objects
  required:    true
  nullable:    false
  description: >
    The decision record entries. One entry per action item from the
    SysAdmin briefing. No action items may be silently dropped; every
    action item from briefing_ref must have a corresponding directive
    entry with an explicit decision.
  item_schema:
    id:
      type:        string
      format:      D-{NN} (sequential two-digit number from D-01)
      required:    true
    briefing_item_ref:
      type:        string
      description: AI id from the source SysAdmin briefing artefact.
      required:    true
    decision:
      type:        enum
      values:      [accepted, deferred, rejected]
      required:    true
    rationale:
      type:        string
      description: >
        Required for all decisions. Accepted entries state what will be
        done and by whom. Deferred entries state the rationale for
        deferral and expected revisit cycle. Rejected entries state why
        the action item is not being taken.
      required:    true
    assigned_to:
      type:        string or null
      description: >
        agent:[slug] or sysadmin:[role] responsible for implementation.
        Must be non-null when decision == accepted.
      required:    true
      nullable:    true
    priority:
      type:        enum
      values:      [low, medium, high, immediate]
      description: Implementation priority. Must be non-null when
                   decision == accepted.
      nullable:    true
    target_completion_cycle_id:
      type:        string or null
      format:      CYCLE-{YYYY}-{NNNN}
      description: >
        The CI cycle by which this directive should be implemented.
        Must be non-null when decision == accepted.
      nullable:    true
    defer_reason:
      type:        string or null
      description: >
        Required and non-null when decision == deferred. States the
        reason for deferral.
      nullable:    true
    revisit_cycle_id:
      type:        string or null
      format:      CYCLE-{YYYY}-{NNNN}
      description: >
        The CI cycle in which this deferred item should be re-evaluated.
        Required and non-null when decision == deferred.
      nullable:    true

## REQUIRED_SECTIONS

SUMMARY:
  heading:     ## SUMMARY
  required:    true
  position:    1
  constraints:
    - Exactly 2–3 sentences
    - Must state directive_count, accepted_count, deferred_count,
      and rejected_count
    - Must reference briefing_ref and cycle_id
    - Must state whether any accepted directives are priority: immediate

BRIEFING_REF:
  heading:     ## BRIEFING_REF
  required:    true
  position:    2
  description: >
    Confirms the identity of the SysAdmin briefing artefact that this
    change directive responds to. Enables traceability from directive
    back to the full CI cycle chain.
  required_fields:
    briefing_ref:       SBR ID — confirms FIELD_05
    cycle_id:           Cycle ID — confirms FIELD_02
    briefing_period:    Human-readable label for the CI cycle period
    action_items_presented: Integer count of action items in the briefing
    decision_recorded_at: Timestamp — mirrors issued_at (FIELD_04)

DIRECTIVES:
  heading:     ## DIRECTIVES
  required:    true
  position:    3
  description: >
    All decision entries — accepted and rejected. Deferred entries are
    listed here only with their D-{NN} id and a note pointing to the
    DEFERRED section; full deferred data is in the DEFERRED section.
  entry_schema:
    Accepted entries:
      id:                    D-{NN}
      briefing_item_ref:     AI id
      decision:              accepted
      rationale:             What will be done
      assigned_to:           agent or role (non-null)
      priority:              enum (non-null)
      target_completion_cycle_id: CYCLE ID (non-null)
    Rejected entries:
      id:                    D-{NN}
      briefing_item_ref:     AI id
      decision:              rejected
      rationale:             Why not taken (non-null)
      note:                  assigned_to, priority,
                             target_completion_cycle_id all null
    Deferred entries (abbreviated):
      id:                    D-{NN}
      briefing_item_ref:     AI id
      decision:              deferred
      note:                  "See DEFERRED section for full detail"
  constraints:
    - All accepted and rejected directives fully specified here
    - D ids sequential from D-01 with no gaps
    - Every AI id in the source briefing must appear exactly once

DEFERRED:
  heading:     ## DEFERRED
  required:    true
  position:    4
  description: >
    Full detail for all deferred directives. When deferred_count == 0,
    this section contains the single line: "No items deferred this cycle."
  entry_schema:
    id:               string — D-{NN} matching entry in DIRECTIVES
    briefing_item_ref: string — AI id
    decision:         deferred
    defer_reason:     string — non-null; reason for deferral
    revisit_cycle_id: string — CYCLE-{YYYY}-{NNNN}; non-null
  constraints:
    - Must not be absent even when empty (write the nil statement)
    - Entry count must equal deferred_count

CHANGE_LOG:
  heading:     ## CHANGE_LOG
  required:    true
  position:    last
  format:      "- v[version] | [YYYY-MM-DD] | [agent_id] | [change summary]"
  constraints:
    - Change directives are not revised after issuance; a new CD artefact
      is issued for corrections. This log records issuance only.

## OPTIONAL_SECTIONS

NOTES:
  heading:      ## NOTES
  description:  Non-normative context from the DLM SysAdmin.
  when_to_include: >
    Exceptional circumstances, overrides of agent-recommended priorities,
    or forward planning notes not captured in directive entries.
  position:     After CHANGE_LOG

## VALIDATION_RULES

V01:
  rule:     directive_id is non-null, matches CD-{YYYY}-{NNNN}, and is
            unique for the year.
  test:     >
    directive_id != null AND
    directive_id matches regex ^CD-\d{4}-\d{4}$ AND
    no other change-directive artefact for the same year shares this id.
  checked_by: CI Cycle Orchestrator; Audit Trail Agent

V02:
  rule:     issued_by must be sysadmin:dlm-sysadmin or
            agent:dlm-sysadmin-agent.
  test:     >
    issued_by IN ["sysadmin:dlm-sysadmin", "agent:dlm-sysadmin-agent"]
  checked_by: CI Cycle Orchestrator

V03:
  rule:     briefing_ref must be non-null and match SBR-{YYYY}-{NNNN} format.
  test:     >
    briefing_ref != null AND
    briefing_ref matches regex ^SBR-\d{4}-\d{4}$
  checked_by: Agent Format Validator; CI Cycle Orchestrator

V04:
  rule:     directive_count must equal accepted_count + deferred_count +
            rejected_count, and must equal the actual count of entries
            in the DIRECTIVES section.
  test:     >
    directive_count == accepted_count + deferred_count + rejected_count AND
    directive_count == len(directives entries)
  checked_by: Agent Format Validator; DLM SysAdmin Agent

V05:
  rule:     D-{NN} identifiers in directives must be sequential starting
            from D-01 with no gaps.
  test:     >
    For all i from 1 to directive_count:
      EXISTS D in directives where D.id == "D-" + zero_pad(i, 2)
  checked_by: Agent Format Validator

V06:
  rule:     Accepted directives must have non-null assigned_to, priority,
            and target_completion_cycle_id.
  test:     >
    For all D in directives where D.decision == "accepted":
      D.assigned_to != null AND
      D.priority != null AND
      D.target_completion_cycle_id != null
  checked_by: Agent Format Validator; DLM SysAdmin Agent

V07:
  rule:     Deferred directives must have non-null defer_reason and
            revisit_cycle_id, and must appear in the DEFERRED section
            with full detail.
  test:     >
    For all D in directives where D.decision == "deferred":
      D.defer_reason != null AND
      D.revisit_cycle_id != null AND
      EXISTS entry in DEFERRED where entry.id == D.id
  checked_by: Agent Format Validator; DLM SysAdmin Agent

V08:
  rule:     Every AI id from the source SysAdmin briefing must appear
            exactly once across all directive entries.
  test:     >
    For each AI id in source SBR artefact RECOMMENDATIONS:
      EXISTS exactly one D in directives where D.briefing_item_ref == AI.id
  checked_by: DLM SysAdmin Agent; Agent Format Validator

V09:
  rule:     All five required sections must be present and appear in the
            prescribed order.
  test:     >
    Sections SUMMARY, BRIEFING_REF, DIRECTIVES, DEFERRED, CHANGE_LOG
    all present and in sequence.
  checked_by: Agent Format Validator; Template Validator
